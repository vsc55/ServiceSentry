{# ================================================================ plan2d — deshacer y rehacer en un plano. En un plano todo se guarda al soltar: arrastrar algo un metro de más, girar lo que no era o estirar con el asa equivocada ya está en el servidor cuando uno se da cuenta. Esto lo devuelve, sin saber de qué es el plano. Antes de mandar un cambio (`PUT`) a algo del plano abierto se mira cómo estaba según lo último GUARDADO —una copia tomada al abrir el plano y puesta al día con cada cambio que el servidor acepta—, no según el dibujo, que al soltar ya enseña lo nuevo. Deshacer es mandar lo de antes. Dos cambios seguidos a la misma cosa son uno; lo que se escribe de una vez —mover un grupo— se deshace de una vez (`p2duBatch`); y un todo que se reemplaza entero —los muros de una planta— se apunta con su antes y su después (`p2duPush`). Lo que es de cada plano lo dice su FUENTE (`p2duPrime(ctx, src)`), una a la vez —el plano que está abierto—: items() [[dirección, fila]] — lo que se puede deshacer, por la dirección en que se escribe send(m, url, b) cómo se manda; flush() lo pendiente de escribir, que sale antes rerender() repintar tras deshacer; apply(url, valores) lo que además hay que poner en el dibujo (la caja de un equipo, los muros de una planta) name(e) cómo se llama lo que se desharía, para el título del botón may() si quien mira puede editar; keysActive() si Ctrl+Z es de este plano ahora Y lo que se edita en memoria hasta guardarlo de una vez —una revisión de muros— se apunta como pasos LOCALES (`p2duLocal`): deshacerlos no manda nada. Mientras esa edición está abierta (`p2duLocalOpen`) el historial solo recorre sus pasos —deshacer lo de antes, debajo, cambiaría lo que ella va a guardar—, y al cerrarla (`p2duLocalClose`) sus pasos se van. Lo que NO se deshace: crear y borrar. Borrar ya pregunta, y lo borrado puede llevarse lo que tenía dentro, que no se recupera reenviando su fila. ================================================================ #} //: El historial del plano abierto: lo guardado de cada cosa (`shadow`, por su dirección), lo //: hecho y lo deshecho (`{url, before, after, at}`), de qué plano es y su fuente. const P2DU = {shadow: new Map(), stack: [], redo: [], ctx: '', src: null, busy: false, localOnly: false, markAt: null}; //: Dos cambios a la misma cosa así de seguidos son uno: escribir «1200» en un campo, o las //: flechas del teclado, no deben costar cuatro deshacer. const P2DU_MERGE_MS = 1500; const P2DU_MAX = 100; /** Tomar la foto de lo guardado al abrir un plano (o al volver a leerlo). Otro plano (`ctx`) * es otra historia: lo del anterior se olvida. */ function p2duPrime(ctx, src) { if (ctx !== P2DU.ctx) { P2DU.ctx = ctx; P2DU.stack = []; P2DU.redo = []; // Otro plano: una edición de pasos locales que se quedara abierta no es de éste. P2DU.localOnly = false; P2DU.markAt = null; } P2DU.src = src || P2DU.src; P2DU.shadow.clear(); for (const [url, o] of (P2DU.src ? P2DU.src.items() : [])) P2DU.shadow.set(url, JSON.parse(JSON.stringify(o))); p2duPaint(); } /** Cómo estaba lo que va a cambiar, campo a campo, o `null` si no es algo del plano. */ function p2duBefore(url, cuerpo) { if (P2DU.busy || !cuerpo || typeof cuerpo !== 'object') return null; const sombra = P2DU.shadow.get(url); if (!sombra) return null; const antes = {}; for (const k of Object.keys(cuerpo)) if (k in sombra) antes[k] = sombra[k]; return Object.keys(antes).length ? antes : null; } /** Apuntar un cambio que el servidor aceptó. Lo que no cambió nada no se apunta: un clic que no * movió es un deshacer que no hace nada. */ function p2duRecord(url, antes, cuerpo) { const sombra = P2DU.shadow.get(url); const despues = {}; for (const k of Object.keys(antes)) despues[k] = cuerpo[k]; if (sombra) Object.assign(sombra, despues); if (Object.keys(antes).every(k => JSON.stringify(antes[k]) === JSON.stringify(despues[k]))) return; const ahora = Date.now(); const top = P2DU.stack[P2DU.stack.length - 1]; // Juntar con el de arriba solo si es de la misma tanda: un cambio dentro de un lote // (`p2duMark`) que se juntara con lo de antes del lote se escapaba de él, y deshacer el lote // dejaba esa cosa a medias. const enTanda = P2DU.markAt === null || P2DU.stack.length - 1 >= P2DU.markAt; if (top && top.url === url && ahora - top.at < P2DU_MERGE_MS && enTanda) { for (const k of Object.keys(antes)) if (!(k in top.before)) top.before[k] = antes[k]; Object.assign(top.after, despues); top.at = ahora; } else { P2DU.stack.push({url, before: antes, after: despues, at: ahora}); if (P2DU.stack.length > P2DU_MAX) P2DU.stack.shift(); } P2DU.redo = []; p2duPaint(); } /** Apuntar un todo que se reemplaza, del que se sabe el antes y el después enteros. */ function p2duPush(url, antes, despues) { P2DU.stack.push({url, before: antes, after: despues, at: 0}); if (P2DU.stack.length > P2DU_MAX) P2DU.stack.shift(); P2DU.redo = []; p2duPaint(); } /** Apuntar un paso que no va al servidor, ANTES de hacerlo: `snap()` dice cómo está ahora y * `restore(estado)` lo pone; `nombre`, para el título del botón. Lo de después se toma al * deshacer: es como esté entonces. */ function p2duLocal(snap, restore, nombre) { P2DU.stack.push({local: true, before: snap(), after: null, snap, restore, name: nombre || '', at: 0}); if (P2DU.stack.length > P2DU_MAX) P2DU.stack.shift(); P2DU.redo = []; p2duPaint(); } /** Olvidar el último paso local si no cambió nada: coger algo y soltarlo sin moverlo. */ function p2duLocalUnchanged() { const e = P2DU.stack[P2DU.stack.length - 1]; if (e && e.local && JSON.stringify(e.before) === JSON.stringify(e.snap())) { P2DU.stack.pop(); p2duPaint(); } } /** Abrir y cerrar una edición de pasos locales. */ function p2duLocalOpen() { P2DU.localOnly = true; p2duPaint(); } function p2duLocalClose() { P2DU.localOnly = false; P2DU.stack = P2DU.stack.filter(e => !e.local); P2DU.redo = P2DU.redo.filter(e => !e.local); p2duPaint(); } /** Lo de arriba de una pila, si se puede deshacer (o rehacer) ahora. */ function p2duTop(pila) { const e = pila[pila.length - 1]; return e && (e.local || !P2DU.localOnly) ? e : null; } /** Por dónde va la pila: lo que se apunte desde aquí se podrá juntar con `p2duBatch`. */ function p2duMark() { P2DU.markAt = P2DU.stack.length; return P2DU.markAt; } /** Juntar en una entrada lo apuntado desde `desde`: lo que se escribió de una vez se deshace de * una vez. Con una sola, nada que juntar. */ function p2duBatch(desde) { P2DU.markAt = null; const partes = P2DU.stack.splice(desde); if (partes.length < 2) { P2DU.stack.push(...partes); return p2duPaint(); } P2DU.stack.push({url: partes[0].url, batch: partes, before: {}, after: {}, at: 0}); p2duPaint(); } /** Deshacer (`dir` = 'undo') o rehacer ('redo') lo último. */ async function p2du(dir) { const src = P2DU.src; const de = dir === 'redo' ? P2DU.redo : P2DU.stack; const a = dir === 'redo' ? P2DU.stack : P2DU.redo; const e = p2duTop(de); if (!e) return; if (e.local) { // En memoria: nada que mandar. Lo de ahora queda como lo de después del paso. de.pop(); if (dir === 'redo') e.restore(e.after); else { e.after = e.snap(); e.restore(e.before); } a.push(e); return p2duPaint(); } if (!src) return; de.pop(); // Lo pendiente de escribir sale antes: si no, llegaría DESPUÉS y desharía el deshacer. if (src.flush) await src.flush(); const valores = dir === 'redo' ? e.after : e.before; P2DU.busy = true; let r; try { if (!e.batch) r = await src.send('PUT', e.url, valores); else { // Al deshacer, de lo último a lo primero; al rehacer, en el orden en que se hizo. const partes = dir === 'redo' ? e.batch : e.batch.slice().reverse(); for (const s of partes) { r = await src.send('PUT', s.url, dir === 'redo' ? s.after : s.before); if (!r || !r.ok) break; } } } finally { P2DU.busy = false; } if (!r || !r.ok) { de.push(e); p2duPaint(); return showToast((r && r.error) || t('save_failed'), 'danger'); } p2duDone(e, valores, a); src.rerender(); } /** Lo deshecho (o rehecho) ya está en el servidor: lo mismo en lo guardado y en el dibujo, y la * entrada a la otra pila (`destino`). */ function p2duDone(e, valores, destino) { if (e.batch) { const lado = valores === e.after ? 'after' : 'before'; for (const s of e.batch) p2duSync(s, s[lado]); } else p2duSync(e, valores); e.at = 0; // lo deshecho no se junta con lo siguiente que se haga destino.push(e); p2duPaint(); } /** Poner lo guardado y el dibujo de una cosa como dicen `valores`. */ function p2duSync(e, valores) { const sombra = P2DU.shadow.get(e.url); if (sombra) Object.assign(sombra, JSON.parse(JSON.stringify(valores))); const src = P2DU.src; if (!src) return; for (const [url, o] of src.items()) { if (url === e.url) Object.assign(o, JSON.parse(JSON.stringify(valores))); } if (src.apply) src.apply(e.url, valores); } /** Qué se desharía: lo que diga la fuente, o cuántas cosas si es un lote. */ function p2duWhat(e) { if (!e) return ''; if (e.local) return e.name; const dicho = P2DU.src && P2DU.src.name ? P2DU.src.name(e) : ''; if (dicho) return dicho; if (e.batch) return tf('plan2d_n_things', e.batch.length); const o = ((P2DU.src ? P2DU.src.items() : []).find(([url]) => url === e.url) || [])[1] || {}; return o.name || o.label || o.uid || ''; } /** Los dos botones de la barra, para quien edita. En su propio hueco, que se repinta solo: un * cambio guardado no repinta la pantalla entera, y el botón tiene que encenderse igual. */ function p2duButtons() { if (!P2DU.src || !P2DU.src.may()) return ''; return `${p2duButtonsInner()}`; } function p2duButtonsInner() { const u = p2duTop(P2DU.stack), r = p2duTop(P2DU.redo); const boton = (dir, icono, e, clave) => { const texto = e ? tf(clave, p2duWhat(e)) : t(clave + '_none'); return ``; }; return boton('undo', 'bi-arrow-counterclockwise', u, 'plan2d_undo') + boton('redo', 'bi-arrow-clockwise', r, 'plan2d_redo'); } function p2duPaint() { const el = document.getElementById('p2dUndoBtns'); if (el) el.innerHTML = p2duButtonsInner(); } /** Ctrl+Z, y Ctrl+Y o Ctrl+Mayús+Z, cuando la fuente dice que es suyo (un plano abierto, nada * que lo tome antes) y fuera de un campo de texto: dentro de uno, deshacer es del campo. */ document.addEventListener('keydown', (ev) => { if (!(ev.ctrlKey || ev.metaKey) || !P2DU.src || !P2DU.src.keysActive()) return; if (/^(INPUT|SELECT|TEXTAREA)$/.test((document.activeElement || {}).tagName || '')) return; const k = (ev.key || '').toLowerCase(); if (k === 'z' && !ev.shiftKey) { ev.preventDefault(); p2du('undo'); } else if (k === 'y' || (k === 'z' && ev.shiftKey)) { ev.preventDefault(); p2du('redo'); } });
