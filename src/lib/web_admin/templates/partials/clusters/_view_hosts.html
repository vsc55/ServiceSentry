{# ================================================================ Clusters — by host: which machines are actually carrying the redundancy. Every cluster row looks redundant on its own. The two ways that is false are both facts about the intersection of the rows, which is why no per-cluster view can show them: * one host carrying several clusters. Each of them says "3 members" and they are the same three, so a single reboot takes out all of them at once. Here that host is one line with several clusters on it, and it sorts to the top. * a cluster with one member. A failover pair with nothing to fail over to — counted in the header rather than hidden in a Members column that reads "1" the same way it reads "3". Pivoted this way the subject is the host and the cluster becomes one of its attributes, so there are no per-cluster actions here: those act on a CLUSTER, and this page is not showing clusters. Offering them per row would invite pressing them against the row in front of you. ================================================================ #} function _clViewHosts(rows) { if (!rows.length) return ''; const by = new Map(); // host uid → {name, clusters: [row]} let orphan = 0; // clusters naming no host at all for (const row of rows) { const members = _clMembers(row); if (!members.length) { orphan++; continue; } for (const m of members) { let h = by.get(m.uid); if (!h) { h = { uid: m.uid, name: m.name, clusters: [] }; by.set(m.uid, h); } h.clusters.push(row); } } // Most-loaded host first: the machine several clusters depend on is the one this view // was opened for, and it is the one whose reboot is a decision rather than a chore. const hosts = [...by.values()].sort((a, b) => (b.clusters.length - a.clusters.length) || a.name.localeCompare(b.name)); const alone = rows.filter(_clIsAlone).length; const shared = hosts.filter(h => h.clusters.length > 1).length; const chips = `${_summaryChip('bi-diagram-3', t('cl_count_clusters'), rows.length)} ${_summaryChip('bi-hdd-network', t('cl_count_hosts'), hosts.length)} ${alone ? _summaryChip('bi-exclamation-triangle', t('cl_count_alone'), alone, 'text-bg-warning') : ''} ${shared ? _summaryChip('bi-layers', t('cl_count_shared'), shared) : ''} ${orphan ? _summaryChip('bi-slash-circle', t('cl_count_orphan'), orphan, 'text-bg-warning') : ''}`; return `
${_summaryHeader(chips)}
${hosts.map(_clHostRow).join('')}
${esc(t('col_host_name'))} ${esc(t('cl_col_clusters'))} ${esc(t('cl_col_in_clusters'))}
`; } function _clHostRow(h) { const host = hostsData[h.uid] || {}; return ` ${esc(h.name)} ${host.address ? ` ${esc(host.address)}` : ''} ${host.maintenance ? ` ${esc(t('host_status_maintenance'))}` : ''} ${h.clusters.length} ${h.clusters.map(row => ` ${_clIsAlone(row) ? '' : ''} ${esc(row.name || row.key)}`).join(' ')} `; }
