{# ================================================================ The account's own API tokens — the view registry. The table is a row per token, which is the shape for acting on one. The two other questions it answers badly are the ones you open the list with once there is more than a handful: • **Cards** — a token is a small object with five facts, and four of them are hidden behind column choices by default. A card shows the lot without choosing, which is what you want while deciding whether a name from six months ago is still the pipeline you think it is. • **By permission** — "what can be done with the tokens I have out there" is a question about permissions, not about tokens, and no ordering of a table answers it: the permissions are a LIST inside one cell, so the flag you are hunting is scattered across rows. A block per permission with its tokens underneath is the same data as the answer. Both extra views are drawn through the factory's `cardsBody` hook, so filters, sorting and pagination stay the SAME machinery — no view re-implements them, and none of them can disagree about which rows are on screen. `perms` is a summary: it is handed every row the filters left standing rather than the page, because a count over one page is a statement about the pagination. ================================================================ #} const ACCOUNT_TOKEN_VIEWS = [ { id: 'table', label_key: 'api_token_view_table', icon: 'bi-table', render: null, mode: 'table' }, { id: 'cards', label_key: 'api_token_view_cards', icon: 'bi-grid-3x3-gap', render: '_tokViewCards', mode: 'cards' }, { id: 'perms', label_key: 'api_token_view_perms', icon: 'bi-shield-lock', render: '_tokViewPerms', mode: 'summary' }, ]; const _tokView = createViewState('ss_apitokens_view', ACCOUNT_TOKEN_VIEWS, () => { _persistTableConfig(); _accTokensRender(); }); function setAccountTokensView(id) { _tokView.set(id); } /** A card per token: every fact at once, and the same three actions the row offers. */ function _tokViewCards(uids) { return `
${uids.map(uid => { const tk = _accTokensByUid[uid] || {}; const dead = tk.revoked || tk.expired; const state = tk.revoked ? `${esc(t('api_token_revoked_badge'))}` : (tk.expired ? `${esc(t('api_token_expired_badge'))}` : ''); return `
${esc(tk.name || '')} ${state}
${_accTokenRowActions(uid)}
sst_${esc(tk.token_id || '')}…
${_accTokenPermsHtml(tk)}
${tk.expires_at ? esc(_fmtDateTime(tk.expires_at)) : esc(t('api_token_never'))} ${tk.last_used ? esc(_fmtDateTime(tk.last_used)) : esc(t('api_token_unused'))}
`; }).join('')}
`; } /** A block per permission, with the tokens that carry it underneath. * * Shares its shape with the administrator's view of the same question (see * partials/apitokens/_views.html) rather than reimplementing the grouping: one of them * drawing `'*'` differently from the other would be two answers to "what can run here". */ function _tokViewPerms(uids) { return _tokPermGroups('apitokens-perms', '_accTokensRender', uids.map(uid => _accTokensByUid[uid] || {}), false, _accTokenRowActions); }
