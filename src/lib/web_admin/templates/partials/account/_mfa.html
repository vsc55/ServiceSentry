{# ================================================================ Account — the second factor: setting one up, turning it off, and the recovery codes. Its own partial because it is the one part of this page that does not go through Save. Every action here talks to the server on its own, because each one either mints a secret or spends a code — and a field in the preferences form would have made one button mean "change my language" and "replace my second factor" at the same time. Two ways to enrol are always drawn: the QR, which is what anybody actually uses, and the base32 key beside it. The key is the half somebody can read back and check, and the only one left when the camera will not focus, the phone has no camera, or the square came out wrong — which is a possibility this project takes seriously, because the QR encoder is written here and no test can hold a phone up to a screen. ================================================================ #} /** What the server says about this account's second factor. Held so the card can redraw * after an action without asking again for what it just changed. */ let _accMfa = null; async function _accMfaLoad() { // `apiGet` answers the parsed BODY; `apiPost`/`apiDelete` answer `{status, data}`. Two // shapes in one module, and reading the wrong one here did not fail loudly — it fell to // the default, so an account with a factor was drawn as "not set up" and the button then // POSTed an enrolment the server correctly refused with 409. One mistake, two symptoms, // and neither of them said what was wrong. const body = await apiGet('/api/v1/account/mfa'); _accMfa = body || {enrolled: false, pending: false, recovery_left: 0}; _accMfaDraw(); } function _accMfaDraw() { const body = document.getElementById('accMfaBody'); const state = document.getElementById('accMfaState'); if (!body || !_accMfa) return; state.innerHTML = _accMfa.enrolled ? `${esc(t('mfa_state_on'))}` : `${esc(t('mfa_state_off'))}`; if (!_accMfa.enrolled) { body.innerHTML = `
${esc(t('mfa_desc'))}
`; return; } body.innerHTML = `
${esc(t('mfa_desc'))}
${ esc(t('mfa_recovery_left').replace('{}', _accMfa.recovery_left))}
`; } /** Start an enrolment and show both ways in. * * The secret is drawn from what the server just stored, so the square and the string are * the SAME secret the confirmation will verify — building the URI in the browser would be a * second place that decides what is being enrolled. */ async function _accMfaBegin() { const res = await apiPost('/api/v1/account/mfa/begin', {}); const d = (res && res.data) || {}; if (!d.ok) { showToast(d.error === 'no_key' ? t('mfa_no_key') : t('save_error'), 'danger'); return; } // The SVG comes from the server as markup built only from integers — module coordinates — // so there is nothing in it that came from a user. Empty when the link did not fit, and // then the key on its own is the whole answer rather than a broken box. const square = d.svg ? `
${d.svg}
` : `
${esc(t('mfa_enrol_no_qr'))}
`; showHtmlModal(t('mfa_enrol_title'), `
${esc(t('mfa_enrol_scan'))}
${square}
${esc(t('mfa_enrol_manual'))}
${esc(d.secret_groups || d.secret || '')}
${esc(t('mfa_enrol_confirm'))}
`, 'primary'); } async function _accMfaConfirm() { const code = (document.getElementById('accMfaCode') || {}).value || ''; const res = await apiPost('/api/v1/account/mfa/confirm', {code}); const d = (res && res.data) || {}; if (!d.ok) { showToast(t('mfa_bad_code'), 'danger'); return; } showToast(t('mfa_enrol_done'), 'success'); _accMfaShowCodes(d.recovery, false); _accMfaLoad(); } /** The recovery codes, once. There is no way back from the stored form — a panel that could * print somebody's codes a second time is one where reading the database is enough. */ function _accMfaShowCodes(codes, replaced) { const list = (codes || []).map(c => `
${esc(c)}
`).join(''); showHtmlModal(t('mfa_recovery_title'), `
${esc(t('mfa_recovery_lead'))}${ replaced ? ' ' + esc(t('mfa_recovery_replaced')) : ''}
${list}
`, 'warning'); } /** Both of these ask for a current code, and neither is a formality: a session is exactly * what an attacker has when they have borrowed one, and without the code the first thing * they would do is take the factor off — or mint ten permanent ways back in. */ function _accMfaAskCode(title, variant, onCode) { showHtmlModal(title, `
${esc(t('mfa_ask_code'))}
`, variant); const go = document.getElementById('accMfaAskGo'); if (go) go.onclick = () => onCode((document.getElementById('accMfaAsk') || {}).value || ''); } function _accMfaRegen() { _accMfaAskCode(t('mfa_recovery_new'), 'primary', async code => { const res = await apiPost('/api/v1/account/mfa/recovery', {code}); const d = (res && res.data) || {}; if (!d.ok) { showToast(t('mfa_bad_code'), 'danger'); return; } _accMfaShowCodes(d.recovery, true); _accMfaLoad(); }); } function _accMfaDisable() { _accMfaAskCode(t('mfa_disable'), 'warning', async code => { const res = await apiPost('/api/v1/account/mfa/disable', {code}); const d = (res && res.data) || {}; if (!d.ok) { showToast(t('mfa_bad_code'), 'danger'); return; } showToast(t('mfa_disabled_ok'), 'success'); _accMfaLoad(); }); }
