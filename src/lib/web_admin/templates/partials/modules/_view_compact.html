{# ================================================================ Modules — view: compact cards + full-width editor. The smallest change from the layout this section had, and it fixes the thing that made that one awkward: cards no longer grow. A card is a status tile — state, items, warnings — and editing leaves the grid entirely for a full-width view of one module. Which means the grid never reflows. In the original, opening the third card pushed the fourth to the next row and you lost your place in a list you were reading. The editor is a pane, not a modal: a modal over a grid you cannot use is a page with extra steps, and this one has a back button that says where it goes back to. ================================================================ #} const _MOD_EDIT_KEY = 'ss_modules_editing'; function editModuleFull(name) { localStorage.setItem(_MOD_EDIT_KEY, name || ''); renderModules(); } function _modViewCompact(c, names) { const editing = localStorage.getItem(_MOD_EDIT_KEY) || ''; // Only stay in the editor while the module is still there AND still shown: deleting it, // or filtering it away, would otherwise strand you on a pane for something you cannot see. if (editing && names.includes(editing)) { _modCompactEditor(c, editing); return; } c.innerHTML = `
${names.map(n => _modTile(_modFacts(n))).join('')}
`; } function _modTile(f) { const accent = f.on ? `linear-gradient(90deg,hsl(${f.hue},65%,50%),hsl(${(f.hue + 45) % 360},65%,55%))` : 'linear-gradient(90deg,#94a3b8,#cbd5e1)'; return `
${_modAvatar(f, 32)} ${esc(f.pretty)} ${esc(f.name)} ${f.unavailable ? '' : _modSwitch(f)}
${_modCountBadge(f) || '—'} ${_modWarnBadge(f)} ${f.unavailable ? '' : ``} ${_modDeleteBtn(f)}
`; } function _modCompactEditor(c, name) { const f = _modFacts(name); c.innerHTML = `
${_modAvatar(f, 30)} ${esc(f.pretty)} ${esc(f.name)} ${_modCountBadge(f)} ${_modWarnBadge(f)} ${f.unavailable ? '' : _modSwitch(f)} ${_modDeleteBtn(f)}
`; _modRenderBodyInto(c.querySelector('#modFullBody'), name); }
