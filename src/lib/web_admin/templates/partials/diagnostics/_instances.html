{# ================================================================ Diagnostics — the other processes of this installation. Everything else on this page describes the process that served the request. On a single container that is the whole installation; split into web / worker / syslog / events it is the web admin and nothing else, and the other three are invisible from every screen — while "is that pod on the same build?" is what a support thread opens with. Read from the heartbeat registry, which each service already writes to, and never over HTTP: the standalone services answer no HTTP unless `SS_CONTROL_TOKEN` is set, which is not the default. The card shows the DIFFERENCE, not four copies of the same list. Four containers built from one image carry identical packages, and printing them four times is a screen nobody reads to find the one row that matters. ================================================================ #} /** The other processes, or nothing at all. * * Absent on a single-process install — there is nothing to compare, and a table of one row * saying "same as this process" exists only to be dismissed. */ function _dgInstancesCard(list) { const rows = (list || []).filter(i => !i.is_self); if (!rows.length) return ''; const drifted = rows.filter(i => i.diff && !i.diff.same).length; const unknown = rows.filter(i => !i.known).length; const state = drifted ? `${esc(t('diag_inst_n_drift') .replace('{}', drifted))}` : `${esc(t('diag_inst_all_same'))}`; return _dgCard('bi-diagram-3', t('diag_card_instances'), `
${esc(t('diag_inst_lead'))}
` + `${rows.map(_dgInstanceRow).join('')}
${esc(t('diag_inst_service'))} ${esc(t('diag_inst_where'))} ${esc(t('diag_inst_version'))} ${esc(t('diag_inst_python'))} ${esc(t('diag_inst_packages'))} ${esc(t('diag_dep_vulns'))}
` + (unknown ? `
${esc(t('diag_inst_unknown_note').replace('{}', unknown))}
` : ''), state); } function _dgInstanceRow(inst) { // The version is the first thing to look at and the reason this card exists: four // containers are supposed to be one image, and a tag left behind shows up here before it // shows up as a bug nobody can reproduce. const mine = ((_diagData || {}).runtime || {}).version || ''; const drift = inst.version && mine && inst.version !== mine; return ` ${esc(inst.service)} ${esc(inst.mode)} ${_dgInstState(inst)} ${esc(inst.host || '—')}${ inst.pid ? esc(` · ${inst.pid}`) : ''} ${_dgInstVersion(inst, drift, mine)} ${esc(inst.python || '—')}${ inst.os ? `· ${esc(inst.os)}` : ''} ${_dgInstPackages(inst)} ${_dgInstVulns(inst)}`; } /** Redraw the rows, and only the rows. * * Both remote checks feed cells in here — the release one fills "Version", the dependency * one fills "CVE" — and neither is going to rebuild a card the reader is looking at. Same * rule the dependency card learned the hard way: the answer is a table update, so the update * is the table. */ function _dgRedrawInstances() { const body = document.getElementById('dgInstBody'); if (!body) return; body.innerHTML = (((_diagData || {}).instances) || []) .filter(i => !i.is_self).map(_dgInstanceRow).join(''); } /** How many advisories affect what THAT container runs. * * A pure join against the answer already in hand: the check asked about the union of every * process's packages in one round, and each instance carries its own name/version pairs. No * second call, and none from the container itself — which is the whole point, because it has * no way out to the internet and no HTTP for us to ask on. */ function _dgInstVulns(inst) { if (!_dgRemote || !_dgRemote.vulnsOk || !inst.known) { return `—`; } const total = _dgInstAdvisories(inst).length; if (!total) return `0`; return ``; } /** `[{id, url, severity, package}]` for one instance — deduplicated across its packages. * * By name AND version, because that is what an advisory is about: the same package at two * versions is two different answers, and reading this one off the other is how a container * gets told it is clean because a different one is. */ function _dgInstAdvisories(inst) { const seen = new Map(); (inst.packages || []).forEach(p => { const row = _dgRemote.byPin[`${p.name}@${p.version || ''}`]; (row ? row.vulns || [] : []).forEach(v => { if (!seen.has(v.id)) seen.set(v.id, {...v, package: `${p.name} ${p.version}`}); }); }); return Array.from(seen.values()); } function _dgShowInstanceVulns(service, host, pid) { const inst = _dgFindInstance(service, host, pid); if (!inst) return; showLinksModal(`${service} · ${host}`, _dgInstAdvisories(inst).map(v => ({ label: v.id, url: v.url, note: [v.package, v.severity ? t('diag_sev_' + v.severity) : ''] .filter(Boolean).join(' — ')})), t('diag_inst_vulns_lead')); } /** The code that container is running, and the two different things that can be wrong with it. * * **Different from here** is the one that matters day to day: they are meant to come from one * image, so a tag left behind shows up here before it shows up as a bug nobody can reproduce. * **Behind the published release** is a separate claim and only exists once somebody has * pressed the release check — the whole installation, this process included, can be perfectly * consistent and perfectly out of date. * * Said with words and an icon, never colour alone: this is a table read at a glance. */ function _dgInstVersion(inst, drift, mine) { if (!inst.version) return `—`; const behind = _dgUpdate && (_dgUpdate.byVersion[inst.version] || {}).status === 'newer'; const note = [drift ? t('diag_inst_version_drift').replace('{}', mine) : '', behind ? t('diag_inst_version_behind').replace('{}', _dgUpdate.tag) : ''] .filter(Boolean).join(' · '); if (!note) return `${esc(inst.version)}`; return `${esc(inst.version)}`; } /** Alive / stale / stopped, in the words the Services tab already uses. */ function _dgInstState(inst) { const cls = {alive: 'text-bg-success', stale: 'text-bg-warning', stopped: 'text-bg-secondary'}[inst.state] || 'text-bg-danger'; return `${esc(t('svc_state_' + inst.state) || inst.state)}`; } /** How its packages compare with this process's — the whole point of the row. * * Both answers open. A difference opens the comparison, because the useful question after * "they differ" is always *in what*. And "same as here (42)" opens the list, because it does * not say WHICH 42 — a sentence that asks the reader to take the interesting half on trust * is the thing this page exists not to do. */ function _dgInstPackages(inst) { if (!inst.known) { return `—`; } const at = `'${escAttr(inst.service)}', '${escAttr(inst.host)}', ${Number(inst.pid) || 0}`; if (inst.diff && inst.diff.same) { return ``; } return ``; } /** Everything that process runs, in the modal the panel already has. * * Read out of the answer already on the page: the row was drawn from it, and fetching the * same thing again to explain it is a second chance for the two to disagree. */ function _dgShowInstancePackages(service, host, pid) { const inst = _dgFindInstance(service, host, pid); if (!inst) return; const pairs = (inst.packages || []).map(p => [p.name, _dgInstPackageLine(p)]); // Whether the remote half is in here at all, said once at the top rather than left to be // inferred from a column of bare versions — which is the same claim as "nothing found" // and the mistake this whole card was built to stop making. const lead = t('diag_inst_packages_lead').replace('{}', inst.version || '?') + ' ' + t(_dgRemote ? 'diag_inst_packages_checked' : 'diag_inst_packages_unchecked'); showInfoModal(`${service} · ${host}`, pairs, lead); } /** One package of another container, with everything known about it in one line. * * The version it runs; whether its lock pins it — over there `pip` is not drift and `flask` * is; and, once somebody has pressed the check, the newest published version and how many * advisories affect THAT version. The remote half is read by name AND version: this * container's copy is not this process's, and answering for one with the other is how a * container gets reported clean because a different one is. */ function _dgInstPackageLine(p) { const out = [p.version || '—']; const rem = _dgRemote ? _dgRemote.byPin[`${p.name}@${p.version || ''}`] : null; if (rem && rem.state === 'behind' && rem.latest) { out.push(`→ ${rem.latest}`); } if (rem && rem.vuln_count) { out.push(t('diag_deps_n_vulns_short').replace('{}', rem.vuln_count)); } if (!p.pinned) out.push(t('diag_deps_unpinned_short')); return out.join(' · ').replace(' · →', ' →'); } /** The instance a row belongs to, by service+host+pid. * * Not by index: the card is redrawn whenever the page refreshes, and an index into a list * that was rebuilt points at whatever moved into that slot. */ function _dgFindInstance(service, host, pid) { return ((_diagData || {}).instances || []).find( i => i.service === service && i.host === host && (Number(i.pid) || 0) === pid) || null; } /** What differs, package by package, against this process. * * Identified by service+host+pid rather than by index: the card is redrawn whenever the page * refreshes, and an index into a list that was rebuilt points at whatever moved into that * slot. */ function _dgShowInstanceDiff(service, host, pid) { const inst = _dgFindInstance(service, host, pid); if (!inst || !inst.diff) return; const label = {missing_there: t('diag_inst_absent_there'), missing_here: t('diag_inst_absent_here')}; const pairs = inst.diff.rows.map(r => [ r.name, r.kind === 'version' ? `${r.here} → ${r.there}` : label[r.kind] || r.kind]); showInfoModal(`${service} · ${host}`, pairs, t('diag_inst_diff_lead').replace('{}', inst.version || '?'), 'warning'); }
