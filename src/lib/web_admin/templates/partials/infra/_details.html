${esc(m.label || m.field)}
${_infraValueHtml(m)}
`).join(''); } // `cap` and not `t`: `t` is the panel's translator, and shadowing it inside a function // that also needs a word is a TypeError on the one branch that has a word in it. const u = Number(used.value), cap = Number(total.value); if (!Number.isFinite(u) || !Number.isFinite(cap) || cap <= 0) { return `
${esc(used.label || used.field)}
${_infraValueHtml(used)}
${esc(total.label || total.field)}
${_infraValueHtml(total)}
`; } // Whole percent in the ring: "10.42 %" inside a 52px circle is a number nobody reads, and // the exact figures are right underneath it. ${esc(fu.value)} ${esc(fu.unit)}
${esc('/ ' + ft.value + ' ' + ft.unit)}
${free ? `
${esc(_fmtMeasure(Number(free.value), free.unit).value)} ${esc(_fmtMeasure(Number(free.value), free.unit).unit)} ${esc(t('infra_free'))}
` : ''}
`; } /** Everything the device says about ONE row, wherever it said it. * * A row is a name the equipment chose, and several profiles can be about the same one: a disk * is a row of SYNOLOGY-DISK-MIB (its model, its temperature, its health) AND the group that * eighty rows of SYNOLOGY-SMART-MIB belong to (`row_group`, split from "Drive 1 / Reallocated * Sector Count" by the pattern that table declares). So the match is on either, and what * comes back is the disk's whole story rather than the profile that happened to be open. */ //: The identity facts that are a NAME for a row rather than a description of it. A model or a //: kind is shared by every disk in the box; a bay, a serial or a name is one row's alone. const _INFRA_ID_ROLES = new Set(['bay', 'serial', 'name', 'id', 'wwn']); //: alias → row_key, for the aliases that belong to exactly one row. Rebuilt with the facts. let _infraRowAlias = new Map(); /** Work out every name this device uses for each of its rows. * * Two tables of one device index it by different columns, and there is no rule that says they * agree: SYNOLOGY-DISK-MIB names a disk by `diskName` ("Drive 1") and files its `diskID` * ("Disk 1") beside it, while SYNOLOGY-SMART-MIB names its rows by a column of its own. So * the SMART rows of a disk grouped under a name the disk's own row never had, and clicking * the disk showed its six readings and none of its eighty attributes. * * Built from what the device SAID — the identity facts it reported for each row — and never * from a rule about how the names look. An alias claimed by two rows is dropped: a Synology * with an expansion unit has a "Drive 3" in the box and a "Drive 3" in the shelf, and * attaching one disk's SMART to both is the failure this is trying to avoid, not a * near-miss of it. */ function _infraIndexRowAliases() { const claims = new Map(); // alias → Set(row_key) const claim = (alias, key) => { const a = String(alias || '').trim(); if (!a || a === key) return; if (!claims.has(a)) claims.set(a, new Set()); claims.get(a).add(key); }; const rowKeys = new Set(), nestedOf = new Map(); // row_key → its group, when nested for (const m of (_infraDetail && _infraDetail.metrics) || []) { if (!m.row_key) continue; rowKeys.add(m.row_key); claim(m.row, m.row_key); if (m.row_group) nestedOf.set(m.row_key, m.row_group); } for (const list of _infraRowFacts.values()) for (const a of list) if (_INFRA_ID_ROLES.has(a.key)) claim(a.value, a.row); // The exact join, when the device offers one. But SYNOLOGY-SMART-MIB also reports the serial of the // disk each of its rows belongs to, and so does the disk table: same physical object, same // number, no convention. Any identifying fact does — the rule is "these two rows are about // the same thing because the device gave them the same name for it". const owner = new Map(); // `role\u0000value` → row_key for (const list of _infraRowFacts.values()) for (const a of list) if (_INFRA_ID_ROLES.has(a.key) && !nestedOf.has(a.row)) { const k = `${a.key}\u0000${a.value}`; if (!owner.has(k)) owner.set(k, new Set()); owner.get(k).add(a.row); } for (const list of _infraRowFacts.values()) for (const a of list) { const group = nestedOf.get(a.row); if (!group || !_INFRA_ID_ROLES.has(a.key)) continue; const owners = owner.get(`${a.key}\u0000${a.value}`); if (owners && owners.size === 1) claim(group, [...owners][0]); } _infraRowAlias = new Map(); for (const [alias, keys] of claims) // Claimed by exactly one row, AND not already the name of a row itself. The second // half is the one that bites: with an expansion unit the shelf disk's split name is // "Drive 3", which is the internal disk's whole name — one claimant, and completely // the wrong disk. A name that already belongs to a row is never a nickname for // another. if (keys.size === 1 && !rowKeys.has(alias)) _infraRowAlias.set(alias, [...keys][0]); } /** Everything the device says about ONE row, wherever it said it. */ function _infraRowDetail(key) { const name = String(key || ''); if (!name) return []; const all = (_infraDetail && _infraDetail.metrics) || []; // Matched on `row_key` — the name the DEVICE composed, before the profile's split — and // never on the split half. Looking them up by the split name puts them // back together, which is the bug that split was written to fix. // // A nested row's group is matched through the alias table, because the table it came from // may index the device by a different column than the row's own table does. return all.filter(m => m.row_key === name || m.row_group === name || (m.row_group && _infraRowAlias.get(m.row_group) === name)); } /** How much of it is NOT already on the card. A chevron on a row that opens onto the two * numbers already in front of you is a promise the dialog does not keep. */ function _infraRowDetailCount(row) { return _infraRowDetail(row).filter(m => !m.headline).length; } /** That story, as a dialog: what the row is, what it says now, and everything behind it. * * Two panes, because a disk answers two different kinds of thing and one list of ninety rows * buries the six that matter. ${_infraRowList(mine, key)}
${_infraNestedTable(nested)}
` : facts + _infraRowList(mine, key); showHtmlModal(String(row), body, 'primary'); } /** What the nested rows are called — the profile's own name for them ("SMART"), not ours. */ function _infraNestedLabel(nested) { const m = nested[0] || {}; return m.source_short || _infraShortLabel(m.source_label || m.source || ''); } /** The nested rows, keyed by the name the device gave each one. */ function _infraNestedRows(nested) { const rows = new Map(); for (const m of nested) { const k = m.row || m.row_key; if (!rows.has(k)) rows.set(k, []); rows.get(k).push(m); } return rows; } /** One line per attribute, one column per reading. * * A SMART attribute is a name and four numbers — current, worst, threshold, raw — and it is * read ACROSS: "is the current value near the threshold" is the whole question, and it cannot * be asked of four separate lines. The columns are whatever fields came back, in the order * the profile declared them, so a table with three of them or six works the same. */ function _infraNestedTable(nested) { const rows = _infraNestedRows(nested); const cols = []; for (const m of nested) if (!cols.some(c => c.field === m.field)) cols.push({field: m.field, label: m.label}); return `
${cols.map(c => ``).join('')} ${[...rows.entries()] .sort((a, b) => String(a[0]).localeCompare(String(b[0]))) .map(([name, list]) => ` ${cols.map(c => { const m = list.find(x => x.field === c.field); return ``; }).join('')} `).join('')}

${esc(name)}${m ? `${_infraValueHtml(m)}` : '—'}
`; } /** The row's own readings, one per line. Few enough that a table of columns would be one * column wide. */ function _infraRowList(items, key) { if (!items.length) return ''; // Two columns on a wide dialog rather than one table across the whole width: a full-width // row made the eye travel the entire modal to join "Temperatura" to "40 °C", with nothing // in between. `).join('')}

Repeating the disk * in every line of a dialog titled with it is the one word on the screen that carries no * information. */ function _infraRowWhich(m, key) { const own = m.row_key && m.row_key !== key ? Alphabetical by key is what it was, which put "Bahía" first and //: "Modelo" third for no reason anybody could see — the keys are internal and their alphabet is //: not a fact about equipment. This is: WHAT it is, WHICH one it is, WHERE it sits, and then //: what it is running. The core may decide this because the core is what names these roles. const _INFRA_FACT_ORDER = ['model', 'model_reported', 'vendor', 'vendor_reported', 'serial', 'kind', 'bay', 'node', 'cluster', 'role', 'status', 'standby', 'firmware', 'version', 'built', 'mac', 'name', 'description', 'location', 'contact', 'sysobjectid']; /** Where a fact sits in that order — anything unlisted goes after, alphabetically. */ function _infraFactRank(key) { const i = _INFRA_FACT_ORDER.indexOf(String(key || '')); return i < 0 ? _INFRA_FACT_ORDER.length : i; } /** The row's identifying facts — a disk's model, serial and bay — above its numbers. */ function _infraRowFactsHtml(key) { const facts = []; // The attributes were recorded against the unsplit name too, so this is an equality and // not a prefix test — `startsWith` matched "Drive 3" against "Drive 30". for (const list of _infraRowFacts.values()) for (const a of list) if (a.row === key) facts.push(a); if (!facts.length) return ''; const seen = new Set(); const uniq = facts.filter(a => !seen.has(a.key) && seen.add(a.key)); uniq.sort((a, b) => _infraFactRank(a.key) - _infraFactRank(b.key) || String(a.key).localeCompare(String(b.key))); const words = _infraStateWords(); return `

${esc(_attrLabel(a.key))}
${esc(text)}
