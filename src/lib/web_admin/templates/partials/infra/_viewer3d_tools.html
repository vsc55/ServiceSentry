{# ================================================================ The 3D viewer, continued — drawing only when something changed, all the solid boxes in a handful of draw calls, flying the camera, pointing at a box, and two fingers. Apart from `_viewer3d.html` because that file is the viewer's grammar — the matrices, the cube, the camera, the input — and this is what was added once a whole building was being drawn on it: a building is thousands of boxes, sixty times a second, for a screen that is usually standing still. ================================================================ #} // ── Drawing only when something changed ────────────────────────────────────── // // The viewer drew sixty frames a second whether anybody touched it or not: a pane left open on // a floor kept a graphics card and a laptop's battery busy drawing the same picture. Now each // frame compares what the picture depends on — the camera, the canvas size, the scene's version, // whether the names show — with the last one drawn, and after a third of a second of nothing // changing the loop stops altogether. Anything that changes the picture wakes it. /** Start the loop if it is stopped. Cheap: called on every input and every rebuild. */ function _ss3dWake(V) { if (!V || !V.gl) return; V.idle = 0; if (!V.raf) V.raf = requestAnimationFrame(() => _ss3dLoop(V)); } /** Wake every viewer on the page: the window changed size, full screen came or went. */ function _ss3dWakeAll() { for (const V of _SS3D.values()) { V.labelsKey = ''; _ss3dWake(V); } } window.addEventListener('resize', _ss3dWakeAll); document.addEventListener('fullscreenchange', _ss3dWakeAll); /** What the picture depends on, as one string. */ function _ss3dFirma(V) { const c = V.cam || {}; return [(c.mira || []).join(','), c.radio, c.theta, c.phi, !!c.orto, V.cv.clientWidth, V.cv.clientHeight, (typeof devicePixelRatio === 'number' ? devicePixelRatio : 1), V.ver || 0, ss3dShowLabels].join('|'); } function _ss3dLoop(V) { V.raf = 0; if (!V.gl) return; _ss3dAnimate(V); const firma = _ss3dFirma(V); if (firma !== V.firma) { V.firma = firma; V.idle = 0; ss3dDraw(V); } else { V.idle = (V.idle || 0) + 1; // The picture has settled: whoever mounted the viewer may want to know where the camera // came to rest — to keep it in the address, so a reload opens the same view. if (V.idle === 20 && typeof (V.opts || {}).onRest === 'function') { try { V.opts.onRest(V); } catch (_) { /* a listener's fault is not the viewer's */ } } } // A few more frames after the last change, then nothing until something wakes it. if (V.anim || V.idle < 20) V.raf = requestAnimationFrame(() => _ss3dLoop(V)); } /** Something in the scene changed that the camera does not show: draw again. */ function _ss3dTouched(V) { V.ver = (V.ver || 0) + 1; _ss3dWake(V); } // ── All the solid boxes in a few draw calls ────────────────────────────────── // // One draw call per box was fine for a room — two hundred — and not for a building: three // thousand boxes is three thousand calls a frame, each setting its matrix and colour. The solid // ones never change while a scene is shown, so they are baked once per scene into big buffers // with their corners already in place and their colour on every vertex, and drawn in a handful // of calls. Walls and glass stay one by one: they are few, and their opacity follows the camera. // // And their edges, as lines a shade darker than the box: two racks side by side of the same // colour read as one long block, and a dark rack against a dark floor as nothing. const _SS3D_BVS = ` attribute vec3 aPos; attribute vec3 aNor; attribute vec3 aCol; uniform mat4 uMVP; varying vec3 vNor; varying vec3 vCol; void main(){ vNor = aNor; vCol = aCol; gl_Position = uMVP * vec4(aPos, 1.0); }`; // The same three lights as the one-by-one program, so a batched box and a box drawn alone look // the same. `uFlat` draws the edges in their colour, unlit. const _SS3D_BFS = ` precision mediump float; uniform float uFlat; varying vec3 vNor; varying vec3 vCol; void main(){ if (uFlat > 0.5) { gl_FragColor = vec4(vCol, 1.0); return; } vec3 n = normalize(vNor); float key = max(dot(n, normalize(vec3(0.45, 0.85, 0.30))), 0.0); float fill = max(dot(n, normalize(vec3(-0.60, 0.35, -0.50))), 0.0); float sky = 0.5 + 0.5 * n.y; float l = 0.32 + 0.22 * sky + 0.55 * key + 0.20 * fill; gl_FragColor = vec4(min(vCol * l, vec3(1.0)), 1.0); }`; //: The twelve edges of the unit cube, as pairs of corners. const _SS3D_EDGES = (() => { const c = []; for (let i = 0; i < 8; i++) c.push([i & 1, (i >> 1) & 1, (i >> 2) & 1]); const out = []; for (let a = 0; a < 8; a++) for (let b = a + 1; b < 8; b++) { const d = Math.abs(c[a][0] - c[b][0]) + Math.abs(c[a][1] - c[b][1]) + Math.abs(c[a][2] - c[b][2]); if (d === 1) out.push(c[a], c[b]); } return out; })(); } /** The solid boxes as arrays, ready for buffers: triangles (position, normal, colour, index) * and edges (position, colour), in batches. for (let v = 0; v < 24; v++) { const p = _ss3dAt(m, _SS3D_CUBE.pos[v * 3], _SS3D_CUBE.pos[v * 3 + 1], _SS3D_CUBE.pos[v * 3 + 2]); const nx = _SS3D_CUBE.nor[v * 3], ny = _SS3D_CUBE.nor[v * 3 + 1], nz = _SS3D_CUBE.nor[v * 3 + 2]; lc.set(oscuro, o); ln.set([0, 1, 0], o); }); }); out.lines.push({pos: lp, nor: ln, col: lc, count: conBorde.length * 24}); } } return out; } function _ss3dBatchProgram(gl) { const p = _ss3dCompile(gl, _SS3D_BVS, _SS3D_BFS); V.batch.tris.concat(V.batch.lines) : [])) for (const k of ['pos', 'nor', 'col', 'idx']) if (b[k]) gl.deleteBuffer(b[k]); V.batch = null; if (!V.bprog) return; const a = ss3dBatchArrays(V.escena.opacos || []); } /** Draw the baked boxes and their edges. `false` when there is nothing baked (no program): the * caller then draws them one by one, as before. */ function _ss3dBatchDraw(V, vp) { const gl = V.gl, B = V.bprog; if (!B || !V.batch) return false; gl.useProgram(B.prog); gl.uniformMatrix4fv(B.loc.mvp, false, vp); const atar = (b) => { gl.bindBuffer(gl.ARRAY_BUFFER, b.pos); gl.enableVertexAttribArray(B.loc.pos); gl.vertexAttribPointer(B.loc.pos, 3, gl.FLOAT, false, 0, 0); gl.bindBuffer(gl.ARRAY_BUFFER, b.nor); gl.enableVertexAttribArray(B.loc.nor); gl.vertexAttribPointer(B.loc.nor, 3, gl.FLOAT, false, 0, 0); gl.bindBuffer(gl.ARRAY_BUFFER, b.col); gl.enableVertexAttribArray(B.loc.col); gl.vertexAttribPointer(B.loc.col, 3, gl.FLOAT, false, 0, 0); }; gl.depthMask(true); // The faces pushed back a hair, so the edges drawn on them win the depth test. gl.enable(gl.POLYGON_OFFSET_FILL); gl.polygonOffset(1, 1); gl.uniform1f(B.loc.flat, 0); for (const t of V.batch.tris) { atar(t); gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, t.idx); gl.drawElements(gl.TRIANGLES, t.count, gl.UNSIGNED_SHORT, 0); } gl.disable(gl.POLYGON_OFFSET_FILL); gl.uniform1f(B.loc.flat, 1); for (const l of V.batch.lines) { atar(l); gl.drawArrays(gl.LINES, 0, l.count); } // The colour attribute off again: left on, pointing at this buffer, the next program's draw // would read past its end. if (B.loc.col >= 0) gl.disableVertexAttribArray(B.loc.col); return true; } // ── Flying the camera ──────────────────────────────────────────────────────── /** Move the camera to `dest` (any of `mira`, `radio`, `theta`, `phi`) over `ms` milliseconds, * easing out.
