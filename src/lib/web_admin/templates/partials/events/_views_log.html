{# ================================================================ Events — the three notification-log views that are not the table. * timeline: the log read as a log — the day stated once, the clock down the left, one sentence per send. Same rows, same order, same page as the table. * by rule: which rule is noisy and which one is failing. The flat list contains this and never states it: forty lines from one rule and one line each from forty rules look the same until they are counted. * by channel: whether a transport is broken. "Email has been failing since 10:00" is a sentence the per-line status column cannot say, because it is about the pattern. The last two are summaries: everything the filters left standing, no pagination — their counts are the point, and a count over one page would change as you paged through. ================================================================ #} // ── Timeline ───────────────────────────────────────────────────────────────── function _evlViewTimeline(rows) { if (!rows.length) return ''; // Day headings follow the order the rows arrive in — the sort control decides the // direction, and re-sorting here would silently override it. let out = '', day = null; for (const e of rows) { const k = _evDayKey(e.ts); if (k !== day) { day = k; out += `
${esc(_evDayLabel(e.ts))}
`; } out += `
${esc(_evClock(e.ts))} ${_evStatusBadge(e.ok)} ${esc(e.rule_name || '—')} ${esc(t('event_src_' + (e.source || 'audit')))} ${_evLogChannels(e).map(c => _evChannelPill(c, 0)).join(' ')} ${e.message ? `${esc(e.message)}` : ''}
`; } return `
${out}
`; } // ── By rule ────────────────────────────────────────────────────────────────── function _evlViewByRule(rows) { if (!rows.length) return ''; const by = new Map(); for (const e of rows) { const k = e.rule_name || '—'; let a = by.get(k); if (!a) { a = { name: k, n: 0, bad: 0, last: 0, lastOk: true, chans: new Set() }; by.set(k, a); } a.n++; if (!e.ok) a.bad++; if ((e.ts || 0) > a.last) { a.last = e.ts || 0; a.lastOk = !!e.ok; } _evLogChannels(e).forEach(c => a.chans.add(c)); } // Failing first, then noisiest: the ones that need a decision above the ones that only // need reading. const list = [...by.values()].sort((x, y) => (y.bad - x.bad) || (y.n - x.n) || x.name.localeCompare(y.name)); const failing = list.filter(r => r.bad).length; return _evlSummaryTable( `${_summaryChip('bi-send', t('evl_count_sends'), rows.length)} ${_summaryChip('bi-bell', t('evl_count_rules'), list.length)} ${failing ? _summaryChip('bi-x-octagon', t('evl_count_failing'), failing, 'text-bg-danger') : ''}`, t('event_name'), false, list.map(a => ` ${esc(a.name)} ${_evlSummaryCells(a, false)} `).join('')); } // ── By channel ─────────────────────────────────────────────────────────────── function _evlViewByChannel(rows) { if (!rows.length) return ''; const by = new Map(); for (const e of rows) { // A line that fanned out to two channels counts under both: this view answers "is // THIS transport working", and that send did travel over both of them. for (const c of _evLogChannels(e)) { let a = by.get(c); if (!a) { a = { name: c, n: 0, bad: 0, last: 0, lastOk: true, chans: new Set() }; by.set(c, a); } a.n++; if (!e.ok) a.bad++; if ((e.ts || 0) > a.last) { a.last = e.ts || 0; a.lastOk = !!e.ok; } } } const list = [...by.values()].sort((x, y) => (y.bad - x.bad) || (y.n - x.n) || x.name.localeCompare(y.name)); const failing = list.filter(c => c.bad).length; return _evlSummaryTable( `${_summaryChip('bi-send', t('evl_count_sends'), rows.length)} ${_summaryChip('bi-broadcast', t('evr_count_channels'), list.length)} ${failing ? _summaryChip('bi-x-octagon', t('evl_count_failing'), failing, 'text-bg-danger') : ''}`, t('event_channels'), true, list.map(a => ` ${esc(_evChannelLabel(a.name))} ${_evlSummaryCells(a, true)} `).join('')); } /** The four columns both summaries share, so "failures" means the same thing in each. The * last send carries its own outcome: 12 failures out of 300 that ended green is a transport * that recovered, and the same numbers ending red is one that is down right now. */ function _evlSummaryCells(a, hideChannels) { return `${a.n} ${a.bad ? `${a.bad}` : '—'} ${hideChannels ? '' : `${ [...a.chans].map(c => _evChannelPill(c, 0)).join(' ') || '—'}`} ${a.last ? `${_evStatusBadge(a.lastOk)} ${esc(_evWhen(a.last))}` : '—'}`; } /** The shell both summaries share. `hideChannels` is passed rather than inferred from the * first column's label: deciding a column by comparing translated text would make the table * lose a column in whichever language happened to translate two headers alike. */ function _evlSummaryTable(chips, firstCol, hideChannels, body) { return `
${_summaryHeader(chips)}
${hideChannels ? '' : ``} ${body}
${esc(firstCol)} ${esc(t('evl_col_sends'))} ${esc(t('evl_col_failures'))}${esc(t('event_channels'))}${esc(t('evl_col_last'))}
`; }
