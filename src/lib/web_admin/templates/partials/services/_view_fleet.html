{# ================================================================ Services — fleet: one row per INSTANCE, across every service. The pivot the card grid cannot do. Today an instance is only visible inside the card of the service it belongs to, so the fleet can be read one service at a time and never as a whole — which hides exactly the two failures a multi-container install has and a single-container one does not: * a follower that stopped reporting while the leader carries on, so the service still says RUNNING and the redundancy is quietly gone; * a container left behind on an older version, which no per-service card can show because the drift is only visible when the versions sit next to each other. So this view puts the instance in the subject position and the service becomes one of its attributes. It is a reading surface: no start/stop here, because those act on a SERVICE and this page is not showing services — offering them per row would invite pressing them against the row you happen to be looking at. ================================================================ #} function _svcViewFleet(data, c) { const all = _svcAllInstances(data); if (!all.length) { c.innerHTML = `
${esc(t('svc_no_instances'))}
`; return; } // Sorted by STATE first, then by identity — never by the heartbeat. // // The first attempt sorted by newest heartbeat, which reads well and is unusable: the // timestamp is the most volatile field in the row, so every poll re-shuffled instances // that tick at similar intervals and the table moved under the cursor. A list that // reorders itself on a timer cannot be read, and cannot be clicked. // // State changes rarely and only when something happened, so an instance jumping to the // top IS the news rather than noise. Below that the order is identity — service, then // host — which does not move at all, so a row stays where you left it between refreshes. const rows = all.slice().sort((a, b) => (_svcRank(b.inst.derived_state) - _svcRank(a.inst.derived_state)) || (a.service.title || '').localeCompare(b.service.title || '') || String(a.inst.host || '').localeCompare(String(b.inst.host || ''))); // Version drift is a fact about the FLEET, not about any one row, so it is computed // across all of them and marked where it shows. With one version there is nothing to say. const versions = new Set(rows.map(r => r.inst.version).filter(Boolean)); const drift = versions.size > 1; c.innerHTML = `
${rows.map(r => _svcFleetRow(r, drift)).join('')}
${esc(t('svc_col_host'))} ${esc(t('svc_col_service'))} ${esc(t('svc_col_state'))} ${esc(t('svc_col_role'))} ${esc(t('svc_col_mode'))} ${esc(t('svc_col_version'))} ${esc(t('svc_col_last'))}
${drift ? `
${esc(t('svc_version_drift'))}
` : ''}
`; } function _svcFleetRow(r, drift) { const inst = r.inst; const when = inst.last_cycle_at || inst.last_seen; // Role: only leader-gated services publish it. Absent is not "standby" — an // active-active service has no leader at all, and saying "standby" would invent a // hierarchy it does not have. const lv = inst.detail && inst.detail.leader; const role = (typeof lv === 'boolean') ? (lv ? `${esc(t('svc_leader'))}` : `${esc(t('svc_standby'))}`) : '—'; const modeK = inst.mode === 'embedded' ? 'svc_mode_embedded' : 'svc_mode_container'; const modeI = inst.mode === 'embedded' ? 'bi-box-seam' : 'bi-hdd-network'; const ver = inst.version ? `v${esc(inst.version)}` : '—'; const self = inst.is_self ? ` ${esc(t('svc_self_tag'))}` : ''; return ` ${esc(inst.host || '—')}${self} ${esc(r.service.title)} ${_svcStateBadge(inst.derived_state)} ${role} ${esc(t(modeK))} ${ver} ${esc(_svcSeenText(when))} `; }
