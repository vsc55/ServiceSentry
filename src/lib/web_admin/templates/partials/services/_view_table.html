{# ================================================================ Services — table: one row per service. The fastest answer to "what is running", and the one that survives growth: the card grid is four columns of tall cards, so a deployment with a dozen services becomes a scroll. Here every service is one line and the whole control plane is visible at once. It gives up the instance detail on purpose — that is what the fleet view is for — and keeps the two things a control surface must never make you hunt for: the state, and the buttons. ================================================================ #} function _svcViewTable(data, c) { const rows = _svcAllMeta(data); if (!rows.length) { c.innerHTML = `

${esc(t('svc_none'))}

`; return; } c.innerHTML = `
${rows.map(_svcTableRow).join('')}
${esc(t('svc_col_service'))} ${esc(t('svc_col_state'))} ${esc(t('svc_instances'))} ${esc(t('svc_col_last'))} ${esc(t('svc_col_detail'))} ${esc(t('svc_col_actions'))}
`; } function _svcTableRow(meta) { // The detail chips a service publishes, kept to the first few by the shared `_chipList`: // a table row that wrapped to three lines would stop being a table row. The card view // shows all of them. const chips = _chipList(meta.detail || [], r => ` ${esc(t(r.label_key))} ${_svcDetailValue(r)}`, 3, ''); const n = (meta.instances || []).length; return `
${_svcAvatar(meta, _SVC_STATE_CLS[meta.state] || 'secondary', false)} ${esc(meta.title)}
${_svcStateBadge(meta.state)} ${n ? n : '—'} ${esc(_svcSeenText(_svcLastSeen(meta)))}
${chips}
${_svcActionsHtml(meta, true)} `; }
