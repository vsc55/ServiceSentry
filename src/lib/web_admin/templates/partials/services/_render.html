// ── Services tab ────────────────────────────────────────────────────────────── // Central view of the background services (scheduler / syslog / worker / DB) with // start/stop for the ones this process hosts. The // buttons beside this menu (Start/Stop) carry icons, so text-only entries read as unfinished. // run_now deliberately avoids the play glyph Start uses — it runs one cycle now, it does not // start the service, and two buttons a centimetre apart should not claim the same action. const _SVC_CMD_ICON = { run_now: 'bi-lightning-charge', reload: 'bi-arrow-clockwise', clear_status: 'bi-eraser', prune: 'bi-trash3', }; // Commands that DELETE something and cannot be undone. They sit in the same dropdown as // Reload — one keystroke apart, no gap, no different colour — so the only thing between a // misplaced click and gone data is being asked first. Declared as data next to the icons so // a new destructive command is one entry away from being guarded, not one review away. const _SVC_CMD_DESTRUCTIVE = new Set(['clear_status', 'prune']); let _svcPollTimer = null; function _svcStateLabel(state) { return t('svc_state_' + state) || state; } function _svcAgo(ts) { if (!ts) return '—'; const secs = Math.max(0, Math.floor(Date.now() / 1000 - ts)); if (secs < 60) return t('svc_just_now'); if (secs < 3600) return Math.floor(secs / 60) + ' min'; if (secs < 86400) return Math.floor(secs / 3600) + ' h'; return Math.floor(secs / 86400) + ' d'; } function _svcIn(secs) { if (secs == null) return '—'; if (secs <= 0) return t('svc_now'); if (secs < 60) return secs + ' s'; if (secs < 3600) return Math.floor(secs / 60) + ' min'; return Math.floor(secs / 3600) + ' h'; } // Formatted value of a backend detail descriptor {label_key, value|value_key, fmt}: // value_key → an i18n value (e.g. "Embedded"); fmt 'ago'/'in' → relative time; // otherwise the value verbatim (escaped). Returns '—' for empty. function _svcDetailValue(r) { if (r.value_key) return esc(t(r.value_key)); if (r.fmt === 'ago') return _svcAgo(r.value); if (r.fmt === 'in') return _svcIn(r.value); return esc(r.value == null ? '—' : String(r.value)); } // Service icon in a tinted square (state colour). Size/layout live in CSS // (.ss-svc-av[-lg]); only the colour + icon are dynamic here. function _svcAvatar(meta, cls, big) { return ``; } // One live instance (heartbeat), two lines: state + host (+ leader) on top, a muted // meta line (this process · mode · version · age) below. Shows remote replicas in // another pod, not only the one this process hosts. function _svcInstanceRow(inst) { const cls = _SVC_STATE_CLS[inst.derived_state] || 'secondary'; const host = esc(inst.host || '—'); const modeK = inst.mode === 'embedded' ? 'svc_mode_embedded' : 'svc_mode_container'; const when = inst.last_cycle_at ? _svcAgo(inst.last_cycle_at) : _svcAgo(inst.last_seen); // Leader / standby badge for single-owner services (the heartbeat detail // carries `leader` only for leader-gated services; absent for active-active). let lead = ''; const lv = inst.detail && inst.detail.leader; if (typeof lv === 'boolean') { lead = lv ? `${esc(t('svc_leader'))}` : `${esc(t('svc_standby'))}`; } // Meta as small pill chips (consistent with the card details) instead of a // flat dot-separated line: this process / mode / version / age. const metaChip = (icon, text, cls) => `${text}`; const metaChips = []; if (inst.is_self) metaChips.push(metaChip('bi-pc-display', esc(t('svc_self_tag')), 'bg-primary-subtle text-primary-emphasis')); metaChips.push(metaChip(inst.mode === 'embedded' ? 'bi-box-seam' : 'bi-hdd-network', esc(t(modeK)))); if (inst.version) metaChips.push(metaChip('bi-tag', 'v' + esc(inst.version))); metaChips.push(metaChip('bi-clock-history', when)); // Where this instance points: shown only for remote instances (embedded ones // share the web admin's DB, already on its Database card). `detail.db` = {main, // syslog?} = {driver, host, name}. let dbLine = ''; const db = inst.detail && inst.detail.db; if (db) { const fmt = x => x ? esc([x.driver, x.host, x.name].filter(Boolean).join(' · ')) : ''; const rows = []; if (db.main) rows.push(`${fmt(db.main)}`); if (db.syslog) rows.push(`${fmt(db.syslog)}`); if (rows.length) dbLine = `
${rows.join('·')}
`; } return `
${esc(_svcStateLabel(inst.derived_state))}${host} ${lead}
${metaChips.join('')}
${dbLine}
`; } // Live instances (heartbeat) block — shared by every card style. function _svcInstancesHtml(meta) { if (!Array.isArray(meta.instances) || !meta.instances.length) return ''; return `
${esc(t('svc_instances'))} (${meta.instances.length})
${meta.instances.map(_svcInstanceRow).join('')}
`; } // Footer (hint · ⋯ commands · start/stop) — shared by every card style. // (No per-card refresh: the Services header already has a global Refresh.) // The buttons a user may press on a service: the commands menu and start/stop. Extracted // from the card footer because every view offers them, and the permission gate must be // asked ONCE — a view that assembled its own buttons would be a view free to offer Stop to // somebody who may not press it. `compact` drops the labels for table-ish rows, where the // icon is enough and the words would set the column width. function _svcActionsHtml(meta, compact) { const canCtl = currentUser.permissions?.includes('services_control'); let primary = ''; if (meta.controllable && canCtl) { const [act, cls, icon, key] = meta.running ? ['stop', 'warning', 'bi-stop-fill', 'svc_stop'] : ['start', 'success', 'bi-play-fill', 'svc_start']; primary = ``; } let cmdMenu = ''; const cmds = _SVC_COMMANDS[meta.key]; if (cmds && canCtl) { const items = cmds.map(a => `
    • `).join(''); cmdMenu = `
      ${items}
    `; } return (cmdMenu || primary) ? `${cmdMenu}${primary}` : ''; } function _svcFooter(meta) { const hint = (!meta.embedded) ? `${esc(t('svc_external_managed'))}` : ''; const right = _svcActionsHtml(meta, false); return (hint || right) ? `
    ${hint} ${right}
    ` : ''; } // Card shell (column + bordered card + accent) shared by the three styles. function _svcShell(meta, cls, body) { return `
    ${body}
    `; } // Render one service card: avatar + title header, a coloured status banner and the // service details as inline chips, then the live instances and the action footer. function _svcCard(meta) { const cls = _SVC_STATE_CLS[meta.state] || 'secondary'; const chips = meta.detail.map(r => ` ${esc(t(r.label_key))} ${_svcDetailValue(r)}`).join(' '); // Solid status banner so the state pops (a faint tint didn't stand out, esp. // gray "stopped"). Yellow/cyan need dark text; the rest read on white. const onLight = (cls === 'warning' || cls === 'info'); const header = `
    ${_svcAvatar(meta, cls, false)}
    ${esc(meta.title)}
    ${esc(_svcStateLabel(meta.state))}
    ${chips ? `
    ${chips}
    ` : ''}`; const body = `${header}
    ${_svcInstancesHtml(meta)}
    ${_svcFooter(meta)}`; return _svcShell(meta, cls, body); } // Build the _svcCard meta from a backend service entry (shared by the full render // and the per-card refresh). function _svcMeta(key, d) { return { key, icon: d.icon || 'bi-gear', title: d.label_key ? t(d.label_key) : key, state: d.state, running: !!d.running, controllable: !!d.controllable, embedded: !!d.embedded, detail: d.detail || [], instances: d.instances || [], }; } // The last payload, so switching view redraws instead of re-fetching: every view reads the // same data, and putting a request on the wire to answer a question about presentation would // also race the poll timer that was about to fetch anyway. let _svcLastData = null; async function renderServices() { const c = document.getElementById('services-container'); if (!c) return; const resp = await apiGet('/api/v1/services'); if (!resp) { _scheduleServicesPoll(); return; } // transient error (e.g. connection lost) → keep the current view, retry later _svcLastData = resp.services || {}; _svcDrawBody(); _scheduleServicesPoll(); } /** Paint the header and hand the body to whichever view is selected. The header is drawn * here rather than by each view so the totals, the switcher and Refresh cannot drift apart * between layouts — and so a view is only ever responsible for the rows. */ function _svcDrawBody() { const c = document.getElementById('services-container'); if (!c || !_svcLastData) return; c.innerHTML = _svcResultsHeader(_svcLastData) + '
    '; const view = SERVICE_VIEWS.find(v => v.id === _svcViewId()) || SERVICE_VIEWS[0]; const draw = window[view.render]; (typeof draw === 'function' ? draw : _svcViewCards)(_svcLastData, c.querySelector('#svc-view-body')); } // ── View: cards ────────────────────────────────────────────────────────────── // What the section had before there was a choice: one card per service, everything about it // in view at once. Kept as the baseline the other three are compared against, and kept in // registry order for the same reason — reordering it would change the thing being compared. function _svcViewCards(data, c) { const cards = _svcAllMeta(data).map(_svcCard); c.innerHTML = `
    ${cards.join('') || `

    ${esc(t('svc_none'))}

    `}
    `; } // Light auto-refresh while the Services tab stays visible. function _scheduleServicesPoll() { if (_svcPollTimer) clearTimeout(_svcPollTimer); _svcPollTimer = setTimeout(() => { if (document.getElementById('tab-services')?.classList.contains('active')) renderServices(); }, 5000); } async function servicesControl(name, action) { const r = await apiPost(`/api/v1/services/${name}/${action}`, {}); if (r && (r.status === 200) && r.data?.ok) { showToast(t('svc_' + action + '_ok'), 'success'); } else { const reason = r?.data?.reason; showToast(reason ? (t('svc_err_' + reason) || reason) : t('svc_action_error'), 'warning'); } renderServices(); } async function servicesCommand(name, action) { // Ask before anything irreversible. The message names the SERVICE, because the same // command destroys different things depending on where it is pressed — Prune under // Syslog drops stored messages, under fail2ban it drops offence counters and the ban // log — and a confirmation that does not say what you are about to lose is a speed bump, // not a safeguard. if (_SVC_CMD_DESTRUCTIVE.has(action)) { const svc = t('svc_' + name) || name; showConfirmModal(tf('svc_cmd_confirm_' + action, svc), () => _servicesCommandRun(name, action)); return; } return _servicesCommandRun(name, action); } async function _servicesCommandRun(name, action) { const r = await apiPost(`/api/v1/services/${name}/command/${action}`, {}); if (r && (r.status === 200) && r.data?.ok) { showToast(t('svc_command_ok'), 'success'); } else { const reason = r?.data?.reason; showToast(reason ? (t('svc_err_' + reason) || reason) : t('svc_command_error'), 'warning'); } renderServices(); }
