{# ================================================================ Syslog — patterns: the same message, counted instead of repeated. Five hundred lines are usually a dozen distinct messages, and the interesting one is often the one that appears twice. Collapsing by "the same message with different numbers in it" turns a wall into a list you can finish, and lifts the rare line out of the noise instead of burying it under the frequent one. Two things it is careful about: * it counts what is LOADED, not what the store holds, and says so on screen. The database can have millions of rows and the browser has one page of them; a bare "128" would be read as the log. The page-size control (up to "all") is how you widen what it covers. * the grouping is a reading aid, not an identity. Numbers, addresses, UUIDs, hex blobs and quoted strings are replaced; words are never touched. Two different messages collapsing into one is a worse failure than two similar ones staying apart, so the rule errs towards keeping them apart. ================================================================ #} function _slViewPatterns(rows) { if (!rows.length) return ''; const by = new Map(); for (const m of rows) { // The severity is part of the key: the same text at `err` and at `info` is two // different events, and merging them would let a warning hide inside chatter. const key = `${m.severity}\u0000${m.app || ''}\u0000${_slPattern(m.message)}`; let g = by.get(key); if (!g) { g = { n: 0, sample: m, devices: new Set(), first: _slWhen(m), last: _slWhen(m), pattern: _slPattern(m.message) }; by.set(key, g); } g.n++; const h = _slHost(m); if (h) g.devices.add(h); const w = _slWhen(m); if (w < g.first) g.first = w; if (w > g.last) g.last = w; } // Most repeated first, then by severity (worst = lowest number) so that among equally // frequent patterns the serious one leads. const list = [...by.values()].sort((a, b) => (b.n - a.n) || (a.sample.severity - b.sample.severity)); const once = list.filter(g => g.n === 1).length; const chips = `${_summaryChip('bi-collection', t('sl_count_patterns'), list.length)} ${once ? _summaryChip('bi-1-circle', t('sl_count_once'), once) : ''}`; return `
${_slLoadedHeader(chips, rows.length)}
${list.map(_slPatternRow).join('')}
${esc(t('sl_col_count'))} ${esc(t('syslog_severity'))} ${esc(t('syslog_app'))} ${esc(t('syslog_message'))} ${esc(t('syslog_host'))} ${esc(t('sl_col_window'))}
`; } function _slPatternRow(g) { const devices = [...g.devices]; // The devices are listed, not just counted: one message from twelve machines and twelve // from one machine are different incidents, and the count alone cannot tell them apart. const hostCell = _chipList(devices, h => _chipText(h), 3); // The sample text is shown, with the pattern underneath only when they differ — on a // message with no variable parts the two lines would be the same sentence twice. const sample = g.sample.message || ''; const showPattern = g.n > 1 && g.pattern !== sample; return ` ${g.n} ${_slSevBadge(g.sample)} ${esc(_slApp(g.sample))} ${esc(sample)} ${showPattern ? `
${esc(g.pattern)}
` : ''} ${hostCell} ${esc(g.first === g.last ? g.last : `${g.first} → ${g.last}`)} `; }
