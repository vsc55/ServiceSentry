{# ================================================================ Roles — usage: who actually holds each role. The Roles table describes a role by what it CAN do — a name, a description, a permission count — and never by who has it. Two questions live in that gap, and both are asked out loud during an access review: * who is an admin here. The answer is spread across the Users table (direct role) and the Groups table (which groups map to it, and who is in them), so today it is assembled by hand. * which roles nobody holds. A role with no users and no groups is configuration that does nothing: harmless, until somebody assigns it years later assuming it was in use and tested. Not an error — which is why it is a marker and not a warning colour. Reach is counted as "users directly + members of the groups that grant it", and a disabled group contributes nothing, because it grants nothing on the server. ================================================================ #} function _rolViewUsage(names) { if (!names.length) return ''; const rows = names.map(r => { const direct = _accUsersOfRole(r); const groups = _accGroupsWithRole(r); // Reach is a UNION: a user who holds the role directly and is also in a group that // grants it must not be counted twice, or a role could report more holders than the // installation has users. const reach = new Set(direct); for (const g of groups) { if (!g.enabled) continue; for (const m of ((groupsData[g.key] || {}).members || [])) reach.add(m); } return { r, direct, groups, reach: reach.size }; }); const unused = rows.filter(x => !x.reach).length; const chips = `${_summaryChip('bi-shield-lock', t('rol_count_roles'), rows.length)} ${unused ? _summaryChip('bi-slash-circle', t('rol_count_unused'), unused, 'text-bg-warning') : ''}`; return `
${_summaryHeader(chips)}
${rows.map(_rolUsageRow).join('')}
${esc(t('col_role_name'))} ${esc(t('rol_col_reach'))} ${esc(t('rol_col_direct'))} ${esc(t('rol_col_groups'))} ${esc(t('col_perm_count'))}
`; } function _rolUsageRow(x) { const rd = rolesData[x.r] || {}; // Names, not a count: which users hold it is the fact you act on. `_chipList` does the // cutting, because a role held by everybody would turn its row into a paragraph. const groupChips = x.groups.length ? x.groups.map(g => ` ${esc(g.name)} ${g.members} ${g.enabled ? '' : ''}`).join(' ') : '—'; return ` ${_accRoleBadge(x.r)} ${rd.builtin ? `` : ''} ${x.reach ? '' : ` ${_accUnusedBadge('rol_unused')}`} ${x.reach || '0'} ${_chipList(x.direct, u => _chipText(u))} ${groupChips} ${(rd.permissions || []).length} `; }
