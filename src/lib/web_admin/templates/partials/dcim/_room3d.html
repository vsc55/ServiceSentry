{# ================================================================ DCIM — la sala levantada, en 3D. El motor —WebGL a pelo, la cámara, los rótulos, el plano en el suelo— es el compartido (`infra/_viewer3d.html`); aquí queda lo que es del inventario: cómo se levanta una sala con sus racks y sus piezas, de qué color es cada cosa —y un rack, del de sus sondas—, y abrir y cerrar el visor de esta sección. El de la sede (`_site3d.html`) usa el mismo visor con otra escena, y levanta cada sala con `_dc3dRoomBoxes`: una sola cuenta de cómo es una sala. ================================================================ #} //: Si el visor está abierto y qué enseña: `{on}`, la sala abierta, o `{on, build, keyOf, //: onPick}`, lo que diga quien lo abrió (el plano de la sede). `null` = apagado. El visor en sí //: —el contexto, el bucle, la cámara— es del motor (`_dc3dView()`). let _dc3d = null; //: A qué altura está el techo cuando nadie lo ha dicho, en milímetros. Lo mismo que declara el //: servidor; escrito aquí porque el dibujo tiene que poder empezar antes de preguntar nada. const _DC3_ROOM_H = 3000; /** Abrir o cerrar el visor. */ function _dcimToggle3d() { if (_dc3d) { _dc3dStop(); renderDcim(); return; } _dc3d = {on: true}; renderDcim(); } //: La caja del visor de esta sección: una sola a la vez, la de la sala o la de la sede. const _DC3D_BOX = 'dc3d-box'; function _dc3dHtml(hint) { return ss3dHtml(_DC3D_BOX, {hint: hint || t('view3d_hint'), extra: _dc3dNorth() !== null ? _dcCompass(0, 'dc3d-compass') : ''}); } /** Montar el visor sobre el lienzo recién pintado (cada repintado pinta otro). La cámara se * conserva mientras se mire lo mismo: la misma sala, o lo que diga `keyOf` en la sede. */ function _dc3dMount() { if (!_dc3d) return; const clave = typeof _dc3d.keyOf === 'function' ? _dc3d.keyOf() : 'room:' + (((_dcimPlan || {}).room || {}).uid || ''); ss3dMount(_DC3D_BOX, {scene: _dc3dScene, key: clave, onPick: _dc3d.onPick, north: _dc3dNorth}); } /** Rehacer la escena sin tocar la cámara: llegaron datos que faltaban (los planos de las otras * plantas) o cambió cómo se miran (las plantas separadas). */ function _dc3dRebuild() { ss3dRebuild(_DC3D_BOX); } function _dc3dStop() { ss3dStop(_DC3D_BOX); _dc3d = null; } //: El rehacer pendiente de `_dc3dSoon`. let _dc3dSoonT = 0; /** Rehacer el 3D en cuanto el plano deje de cambiar. Los planos repintan solo su dibujo —sin * `renderDcim`— al arrastrar, girar, editar en el inspector o deshacer, y el 3D seguía * enseñando lo de antes hasta el siguiente repintado completo: la mesa movida, en su sitio * viejo. No en cada repintado: un arrastre son decenas por segundo, y levantar la escena otras * tantas veces es trabajo que nadie ve. Se espera a que pare y se rehace una vez. */ function _dc3dSoon() { if (!_dc3d) return; clearTimeout(_dc3dSoonT); _dc3dSoonT = setTimeout(_dc3dRebuild, 150); } /** El visor montado, o `null`. */ function _dc3dView() { return ss3dGet(_DC3D_BOX); } //: Colores por tipo, en 0..1 con alfa. El vidrio y el confinamiento van translúcidos porque en //: una sala real se ve a través de ellos, y un pasillo confinado opaco esconde justo la fila de //: racks que se quería mirar. const _DC3_COLOR = { // El muro, claro como una pared, y el rack, oscuro como un armario: eran casi el mismo gris // —0,22 contra 0,20— y un rack contra la pared se confundía con ella. floor: [0.30, 0.32, 0.35, 1], wallroom: [0.50, 0.52, 0.55, 1], ground: [0.24, 0.25, 0.28, 1], tile: [0.40, 0.42, 0.46, 1], rack: [0.13, 0.14, 0.17, 1], aisle: [0.62, 0.68, 1.00, 0.16], zone: [0.35, 0.38, 0.43, 0.35], column: [0.36, 0.37, 0.40, 1], wall: [0.60, 0.66, 1.00, 0.22], door: [0.95, 0.76, 0.05, 1], panel: [0.82, 0.80, 0.74, 1], ups: [0.30, 0.32, 0.37, 1], crac: [0.20, 0.44, 0.50, 1], bench: [0.46, 0.37, 0.27, 1], extinguisher: [0.72, 0.16, 0.16, 1], cabinet: [0.42, 0.44, 0.48, 1], // La franja del delante de una pieza que lo tiene: el azul del tema, como en el plano. front: [0.23, 0.48, 0.92, 1], tray: [0.95, 0.76, 0.05, 0.85], label: [0.35, 0.38, 0.43, 0.5], }; //: Y el color de un rack por cómo está, que es lo que ninguna librería podía dar: sale de las //: sondas. Un armario rojo desde la puerta de la sala es la razón de que esto exista. const _DC3_STATE = { error: [0.86, 0.21, 0.27, 1], warning: [0.95, 0.68, 0.09, 1], ok: [0.15, 0.65, 0.35, 1], }; /** Todo lo que hay que pintar, como una lista de cajas en METROS. * * Se construye una vez por apertura y no en cada fotograma: son doscientas cajas y sesenta * fotogramas por segundo, y recalcularlas es trabajo que no cambia de resultado. La sala * abierta, salvo que quien abrió el visor diga otra cosa (`_dc3d.build`): el plano de la sede * lo abre con una planta, o con el edificio entero. */ function _dc3dScene() { if (_dc3d && typeof _dc3d.build === 'function') return _dc3d.build(); const escena = _dc3dRoomBoxes((_dcimPlan || {}).room || {}, (_dcimPlan || {}).racks || [], _dcpFeatures(), _dcpKinds(), {air: _dcpAir}); // Y el plano de la sala en su suelo, donde lo pone la calibración: unos milímetros por // encima de él, que a la misma altura pelearían por cada píxel. const img = (_dcimPlan || {}).image; if (img && img.data && _dcPlanBg) { const b = _dcpPlanBox(), M = 0.001; escena.planos = [{data: img.data, alpha: b.scaled ? 0.85 : 0.5, m: ss3dModel(b.x * M, 0.01, b.y * M, b.w * M, 1, b.h * M, 0)}]; } return escena; } /** Las cajas de UNA sala, en sus coordenadas: la esquina en (0, 0) y sin girar. * * Una función de los datos y no de la sala abierta, porque la misma sala se pinta sola —su * visor— y dentro de una planta, girada y en su sitio (`ss3dPlace`). Dos cuentas de cómo se * levanta una sala serían dos salas que no se parecen según desde dónde se miren. * * `o`: `tiles` (las líneas de baldosa: en una planta son la mayor parte de las cajas y no se * leen desde lejos), `air` (lo que cuelga), `walls` y `floor` (la zona general de una planta * no tiene ni lo uno ni lo otro: es la planta misma), `floorY` (dónde empieza su suelo; en una * planta, encima del de la planta y no a la misma altura, que pelearían por el mismo píxel) y * `ghost` (una sala que no ha dicho cuánto mide: tenue, para que no parezca medida). */ function _dc3dRoomBoxes(room, racks, features, kinds, o) { o = Object.assign({tiles: true, air: true, walls: true, floor: true, floorY: -0.02, ghost: false}, o || {}); const M = 0.001; // milímetros → metros const W = (Number(room.width_mm) || 0) * M; const D = (Number(room.depth_mm) || 0) * M; const H = _DC3_ROOM_H * M; const cajas = []; const tenue = (c) => (o.ghost ? [c[0], c[1], c[2], Math.min(c[3], 0.3)] : c); const push = (color, x, y, z, w, h, d, rot, lee, pivot) => cajas.push({color, m: ss3dModel(x, y, z, w, h, d, rot, pivot), lee}); // El suelo, del tamaño de la sala si alguien lo dijo y del de lo que hay si no: una sala sin // medidas dibujada sobre la nada deja los racks flotando. const cx = W || 12, cz = D || 8; // Tenue, si la sala no ha dicho cuánto mide, solo lo que la delimita —su suelo y sus muros—: // lo de dentro sí está, y medido. if (o.floor) push(tenue(_DC3_COLOR.floor), 0, o.floorY, 0, cx, -o.floorY, cz, 0, null); // Los cuatro muros, cada uno con el lado de la sala en que queda FUERA: el que se interpone // entre la cámara y la sala se dibuja de cristal (`ss3dWallAlpha`). Antes solo había los // dos del fondo —los cuatro tapaban la vista desde fuera—, y cruzar uno de ellos dejaba la // pantalla en gris: se miraba la sala a través de su muro, por detrás. const muro = (x, z, w, d, lado) => cajas.push({color: tenue(_DC3_COLOR.wallroom), m: ss3dModel(x, 0, z, w, H, d, 0), lee: null, muro: lado}); if (o.walls) { muro(0, -0.06, cx, 0.06, 'z0'); muro(-0.06, 0, 0.06, cz, 'x0'); muro(0, cz, cx, 0.06, 'z1'); muro(cx, 0, 0.06, cz, 'x1'); } // Las baldosas, como líneas en el suelo: sin ellas no hay con qué medir a ojo, y en una // vista en perspectiva «a dos baldosas» es la única distancia que se lee. const baldosa = (Number(room.tile_mm) || 600) * M; if (o.tiles && baldosa > 0.1 && cx / baldosa < 200 && cz / baldosa < 200) { for (let x = baldosa; x < cx - 1e-6; x += baldosa) push(_DC3_COLOR.tile, x - 0.004, 0, 0, 0.008, 0.001, cz, 0, null); for (let z = baldosa; z < cz - 1e-6; z += baldosa) push(_DC3_COLOR.tile, 0, 0, z - 0.004, cx, 0.001, 0.008, 0, null); } for (const r of (racks || [])) { const w = (Number(r.width_mm) || 600) * M, d = (Number(r.depth_mm) || 1000) * M; // Lo alto que es un rack sale de sus U, no de una constante: un armario de 42U y uno de // 12 son dos cosas distintas y en planta se ven iguales. const h = Math.max(0.3, (Number(r.u_height) || 42) * 44.45 * M); const estado = (r.roll || {}).state || ''; // Un rack de pared empieza donde está colgado, no en el suelo. const base = Math.max(0, Number(r.base_mm) || 0) * M; const rx = (Number(r.pos_x) || 0) * M, rz = (Number(r.pos_y) || 0) * M; const rot = Number(r.rotation) || 0; push(_DC3_COLOR.rack, rx, base, rz, w, h, d, rot, {name: r.name || '', kind: t('dcim_rack'), w, d, h, state: estado}); // La franja del frente, que es lo que dice cuál es el pasillo frío. En el MISMO lado que // en planta —el de `pos_y`, donde el plano pinta su franja—: estaba en el de atrás, y el // plano y el 3D decían cosas contrarias sobre hacia dónde sopla el rack. Un poco por // delante de la cara para que no pelee con ella por el mismo píxel, y girando con el // rack, alrededor de SU centro: sobre el suyo propio se iría a otro lado. push(_DC3_STATE[estado] || [0.95, 0.76, 0.05, 1], rx, base + h * 0.02, rz - 0.012, w, h * 0.96, 0.012, rot, null, [rx + w / 2, rz + d / 2]); } for (const f of (features || [])) { const spec = (kinds || {})[f.kind] || {}; if ((spec.layer === 'air') && !o.air) continue; const w = (Number(f.width_mm) || 600) * M, d = (Number(f.depth_mm) || 600) * M; // Lo suyo si lo tiene, lo de su tipo si no: la misma cuenta que el alzado. const z = _dcpPieceZ(f, kinds || {}); const h = Math.max(0.01, z.h * M); const base = z.base * M; const fx = (Number(f.pos_x) || 0) * M, fz = (Number(f.pos_y) || 0) * M; const rot = Number(f.rotation) || 0; const lee = {name: f.label || '', kind: t('dcim_kind_' + f.kind), w, d, h, state: ''}; const color = _DC3_COLOR[f.kind] || [0.3, 0.32, 0.36, 1]; if (f.kind === 'cabinet') { _dc3dCabinet(push, color, fx, base, fz, w, h, d, rot, Math.max(1, Number(f.shelves) || 1), lee); } else { push(color, fx, base, fz, w, h, d, rot, lee); // El delante, como en un rack: una franja en la cara del lado de `pos_y`, girando // con la pieza alrededor de SU centro. if (spec.front) push(_DC3_COLOR.front, fx, base + h * 0.02, fz - 0.012, w, h * 0.96, 0.012, rot, null, [fx + w / 2, fz + d / 2]); } } return {cajas, W: cx, D: cz, H, planos: []}; } /** Un armario abierto por delante —el lado de `pos_y`—: laterales, fondo, techo y suelo, y * una balda por cada estantería. Macizo era un bloque que no dejaba ver que tiene baldas ni * cuántas, que es lo que se le pregunta. Todas las tablas giran alrededor del centro del * armario, no del suyo: girando cada una sobre sí misma se desmontaría. */ function _dc3dCabinet(push, color, x, y, z, w, h, d, rot, baldas, lee) { const g = Math.min(0.025, w / 10, d / 10, h / 10); // el grueso de una tabla const centro = [x + w / 2, z + d / 2]; const tabla = (c, tx, ty, tz, tw, th, td, l) => push(c, tx, ty, tz, tw, th, td, rot, l || null, centro); const claro = [Math.min(1, color[0] * 1.35), Math.min(1, color[1] * 1.35), Math.min(1, color[2] * 1.35), 1]; tabla(color, x, y, z + d - g, w, h, g, lee); // el fondo tabla(color, x, y, z, g, h, d); // los lados tabla(color, x + w - g, y, z, g, h, d); tabla(color, x, y + h - g, z, w, g, d); // el techo for (let k = 0; k < baldas; k++) { // el suelo y las baldas tabla(claro, x + g, y + (h - g) * k / baldas, z, w - 2 * g, g, d - g); } } /** El norte de lo que se mira: el de la sala, o el de la planta abierta. `null` si no se * sabe. */ function _dc3dNorth() { if (_dc3d && typeof _dc3d.build === 'function') return typeof _dcsFloor === 'function' ? _dcNorthOf(_dcsFloor()) : null; return _dcNorthOf((_dcimPlan || {}).planFloor || (_dcimPlan || {}).room); }
