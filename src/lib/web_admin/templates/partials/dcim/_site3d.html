Uno * por planta y una sola vez; al llegar, la escena se rehace sin mover la cámara. */ async function _ds3LoadImages() { if (!_dcsPlan) return; const faltan = _dcsPlan.floors.filter(f => f.plan && f.uid !== _dcsPlan.floor && !(f.uid in _ds3Images)); if (!faltan.length) return; const leidas = await Promise.all(faltan.map(f => _dcsFetchImage(f))); faltan.forEach((f, i) => { _ds3Images[f.uid] = leidas[i]; }); _dc3dRebuild(); } function _ds3Img(floor) { if (!_dcsPlan || !floor) return null; return floor.uid === _dcsPlan.floor ? _dcsPlan.image : (_ds3Images[floor.uid] || null); } /** Lo que hay que pintar: la planta abierta, o el edificio. */ function _ds3Scene() { const vacia = {cajas: [], planos: [], salas: [], W: 12, D: 8, H: 3}; if (!_dcsPlan) return vacia; const actual = _dcsFloor(); if (_ds3Mode !== 'building') return actual ? _ds3FloorScene(actual, _ds3Img(actual)) : vacia; return _ds3BuildingScene(actual); } /** Una planta, en metros y con su suelo en y = 0. * * De abajo arriba: la losa de la planta, el plano del arquitecto encima de ella, el suelo de * cada sala encima del plano —que lo tapa donde hay sala— y lo que hay en cada una. El rayo se corta con el suelo de la * planta y el punto se lleva a las coordenadas de cada sala, giro incluido. */ function _ds3Pick(rayo) { const e = (_dc3d || {}).escena || {}; let mejor = null; for (const s of (e.salas || [])) { if (Math.abs(rayo.d[1]) < 1e-6) continue; const t0 = (s.y - rayo.o[1]) / rayo.d[1]; if (t0 <= 0 || (mejor && t0 >= mejor.t)) continue; const px = rayo.o[0] + rayo.d[0] * t0, pz = rayo.o[2] + rayo.d[2] * t0; const f = s.frame, a = f.rot * Math.PI / 180, c = Math.cos(a), sn = Math.sin(a); const dx = px - (f.x + f.W / 2), dz = pz - (f.z + f.D / 2); const lx = dx * c + dz * sn + f.W / 2, lz = -dx * sn + dz * c + f.D / 2; if (lx >= 0 && lx <= f.W && lz >= 0 && lz <= f.D) mejor = {t: t0, sala: s}; } const el = document.getElementById('dc3d-read'); const ahora = Date.now(); if (!mejor) { _ds3LastPick = null; if (el) el.textContent = t('dcim_3d_site_hint'); return null; } const uid = mejor.sala.uid; if (_ds3LastPick && _ds3LastPick.uid === uid && ahora - _ds3LastPick.t < 450) { _ds3LastPick = null; _dcimOpenPlan(uid); return uid; } _ds3LastPick = {uid, t: ahora}; if (el) el.textContent = tf('dcim_3d_room_pick', mejor.sala.name || uid); return uid; } /** Encima del visor: planta o edificio, y en el edificio, separar las plantas. */ function _ds3Bar() { const modo = (m, icono, texto) => ``; return `

`; }
