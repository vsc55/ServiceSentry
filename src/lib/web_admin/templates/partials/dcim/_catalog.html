    return fuera.sort((a, b) => _DCCAT_PORTS.indexOf(a.fam) - _DCCAT_PORTS.indexOf(b.fam)

}

/** Los puertos de una fila, como una cuenta por familia. Lo guardado es `{familia: {tipo: n}}`
 *  y aquí se suman los tipos: sirve para enseñar «4 interfaces» de un vistazo donde no cabe
 *  más. */
function _dcCatPortCounts(r) {
    let p = (r || {}).ports;
    if (typeof p === 'string') { try { p = JSON.parse(p); } catch (e) { p = {}; } }
    const out = {};
    Object.entries(p || {}).forEach(([familia, tipos]) => {
        out[familia] = Object.values(tipos || {}).reduce((a, b) => a + Number(b || 0), 0);
    });
    return out;
}

async function _dcCatImgUp(face, inp) {
    const file = (inp && inp.files && inp.files[0]) || null;
    if (!file || !_dcCatEditing) return;

    // `fetch` a pelo: lo que va aquí es un formulario con un fichero dentro, y el envoltorio de
    // JSON lo convertiría en la cadena "[object File]". El global sigue poniendo el CSRF.
    let resp;
    try {
        resp = await fetch(`/api/v1/dcim/catalog/${encodeURIComponent(_dcCatEditing.uid)}` +
                           `/image/${encodeURIComponent(face)}`, {method: 'POST', body: fd});
    } catch (e) {
        return showToast(t('save_failed'), 'danger');
    }
    const d = await resp.json().catch(() => ({}));
    if (!resp.ok || d.error) return showToast(d.error || `HTTP ${resp.status}`, 'danger', 8000);
    _dcCatEditing[face + '_image'] = d.image || '';
    showToast(t('saved'), 'success');
    _dcCatFormDraw();
    _dcCatOpen();
}

function _dcPortImport(luego) {
    const caja = document.createElement('input');
    caja.type = 'file';
    caja.accept = '.json,application/json';
    caja.onchange = () => {
        const fichero = (caja.files || [])[0];
        if (!fichero) return;
        const lector = new FileReader();
        lector.onload = async () => {
            let doc = null;
            try { doc = JSON.parse(String(lector.result || '')); }
            catch (e) { return showToast(t('dcim_port_bad'), 'danger'); }
            const r = await _dcimSend('POST', '/api/v1/dcim/import', doc);
            if (!r.ok) return showToast(r.error || t('save_failed'), 'danger');
            showToast(tf('dcim_port_done', Number(r.types_new || 0),
                         Number(r.builds_new || 0),
                         Number(r.types_skipped || 0) + Number(r.builds_skipped || 0)),
                      'success');
            // Una plataforma que no existe aquí no se inventa, y por eso hay que decir cuál:
            // la plantilla entró sin ella y nadie lo notaría hasta abrirla.
            const faltan = r.platforms_missing || [];
            if (faltan.length)
                showToast(tf('dcim_port_no_plat', faltan.join(', ')), 'warning');
            if (typeof luego === 'function') luego();
        };
        lector.readAsText(fichero);
    };
    caja.click();
}

/** El rótulo de un campo **escrito a mano**, con su ayuda si la tiene.
 *
 *  Hermano de `_dcAttrLabel`, que hace lo mismo con los campos que declara el documento de
 *  perfiles: aquellos llevan su ⓘ desde el principio y los escritos en la plantilla no, así que
 *  en el mismo formulario unos campos explican qué son y otros no — y los que no son justo los
 *  que se rellenan el primer día sin tener el chasis delante.
 *
 *  Misma convención —la ayuda es `_tt`— y el icono **solo si existe**: uno al lado de cada
 *  campo, diga algo o no, deja de significar «aquí hay algo que leer». */
function _dcLabel(clave, kAyudaOtra) {
    const texto = t(clave);
    // La ayuda se puede pedir aparte porque el mismo rótulo significa cosas distintas en dos
    // formularios: el «Fondo» de un armario es el de la caja entera y el de una plantilla es lo
    // que ocupa dentro de ella. Un solo texto para los dos sería el de uno, mal, en el otro.
    const kAyuda = kAyudaOtra || (clave + '_tt');
    const ayuda = t(kAyuda);
    return _dcHelp(texto, (!ayuda || ayuda === kAyuda) ? '' : ayuda);
}

/** Un rótulo YA TRADUCIDO con su ayuda al lado.
 *
 *  Aparte de `_dcLabel` porque no todo rótulo viene de una clave: los cuadros de los modales de
 *  cableado y alimentación reciben el texto hecho. Sin esto, ponerles ayuda era copiar aquí el
 *  `` y el icono — y la copia es la que se queda sin el cambio del día que el icono
 *  deje de ser un interrogante. */
function _dcHelp(texto, ayuda) {
    if (!ayuda) return esc(texto);
    return `${esc(texto)}
        `;
}

/** Cómo se llama un atributo o una columna. Con la clave sin traducir como respaldo, por lo mismo
 *  que los puertos: el servidor puede ofrecer uno que esta pantalla todavía no conozca. */
function _dcAttrName(k) {
    // `extra.interface` es la interfaz: el prefijo dice de qué columna salió y no es parte del
    // nombre. Se quita aquí y no al calcular la diferencia, porque allí sí hace falta para que
    // dos atributos de columnas distintas no se pisen.
    const nombre = String(k || '').replace(/^(extra|ports)\./, '');
    const clave = _DC_FIELD_KEY[nombre] || ('dcim_attr_' + nombre);
    const texto = t(clave);
    return (texto && texto !== clave) ? texto : nombre;
}

/** Qué hizo ese cambio. La acción cuando no cambió ningún campo comparable —poner una imagen no
 *  cambia ninguno— y si no, los campos con lo que decían y lo que pasaron a decir. */
function _dcCatHistWhat(h) {
    const cambios = Object.entries(h.changes || {});
    if (!cambios.length)
        return `${
            esc(t('dcim_cat_hist_' + (h.action || 'edit')) || h.action || '')}`;
    return cambios.map(([campo, par]) => `
${
        esc(_dcAttrName(campo))}: ${
        esc(_dcCatHistVal(par[0]))} → ${
        esc(_dcCatHistVal(par[1]))}
`).join('');
}

function _dcCatHistVal(v) {
    if (v === null || v === undefined || v === '') return '—';
    if (typeof v === 'object') return JSON.stringify(v);
    return String(v);
}

/** Los campos que el esquema pide y el formulario no trae de fábrica.
 *
 *  Los de siempre —fabricante, modelo, U, puertos— ya están arriba: repetirlos sería pedirlos
 *  dos veces. Lo que sale aquí es lo que el esquema añade, que es exactamente lo que hace que
 *  traerlos sirva de algo — y lo que un esquema propio existe para pedir.
 *
 *  Todos van a `extra`, y se dice: un campo de ahí se guarda y se enseña, pero no se busca ni se
 *  filtra por él, y quien añada uno esperando lo contrario se llevaría la sorpresa el día que
 *  intente encontrar algo. */
function _dcCatSchemaFields() {
    const e = _dcCatSchema;
    if (!e) return '';
    // Lo que el formulario ya pide por su cuenta. Se nombra aquí y no se deduce: la lista de
    // arriba es fija y corta, y adivinarla sería adivinar.
    const yaEstan = ['manufacturer', 'model', 'slug', 'description', 'part_number', 'u_height',
                     'is_full_depth', 'is_powered', 'airflow', 'subdevice_role',
                     'form_factor', 'width', 'starting_unit', 'outer_width', 'outer_height',
                     'outer_depth', 'outer_unit', 'mounting_depth', 'max_weight',
                     'weight', 'weight_unit'];
    const suyos = (e.fields || []).filter(c => c.target === 'extra'
                                               && yaEstan.indexOf(c.name) < 0);
    if (!suyos.length) return '';
    return `
${esc(tf('dcim_cat_schema_extra', e.name))}

        
${suyos.map(c => {
        const v = _dcCatFormExtra[c.name];
        const valor = v === undefined || v === null ? '' : String(v);
        if ((c.enum || []).length) {
            return ``;
        }
        const tipo = (c.type === 'integer' || c.type === 'number') ? 'number' : 'text';
        if (c.type === 'boolean')
            return ``;
        return ``;
    }).join('')}
`;
}

async function _dcCatEditSave() {
    const f = _dcCatForm;
    if (!String(f.manufacturer || '').trim() || !String(f.model || '').trim())
        return showToast(t('dcim_catalog_need_name'), 'warning');
    // Los puertos, de renglones a la forma que guarda la biblioteca: `{familia: {tipo: n}}`.
    // Dos renglones de la misma familia y el mismo tipo se **suman** en vez de pisarse: quien
    // escribe «2 c14» y luego «2 c14» más quiere cuatro, no dos.
    const ports = {};
    (_dcCatFormPorts || []).forEach(p => {
        const v = Number(p.n || 0);
        if (!(v > 0) || !p.fam) return;
        const tipo = String(p.tipo || '').trim();
        ports[p.fam] = ports[p.fam] || {};
        ports[p.fam][tipo] = (ports[p.fam][tipo] || 0) + v;
    });
    // Y las medidas, sin los huecos que se dejaron vacíos: una medida en blanco no es un cero,
    // es «no lo sé», y guardarla como cero diría que el armario mide nada de fondo.
    const extra = {};
    Object.entries(_dcCatFormExtra).forEach(([k, v]) => {
        if (v === '' || v === null || v === undefined) return;
        const n = Number(v);
        extra[k] = (v !== true && v !== false && String(v).trim() !== '' && !isNaN(n)) ? n : v;
    });
    const cuerpo = {
        manufacturer: f.manufacturer, model: f.model, slug: f.slug,
        description: f.description, part_number: f.part_number,
        tree: f.tree, kind: f.kind, size: f.size || '', url: f.url || '',
        kit_qty: Number(f.kit_qty || 1) || 1, power_type: f.power_type || '',
        u_height: f.tree === 'module-types' ? 0 : f.u,
        is_powered: f.tree === 'rack-types' ? 0 : (f.is_powered ? 1 : 0),
        full_depth: f.full_depth ? 1 : 0,
        airflow: f.airflow, subdevice: f.subdevice,
        ports: ports, extra: extra,
    };
    if (_dcCatCloneFrom) cuerpo.from = _dcCatCloneFrom;
    const r = _dcCatEditing
        ? await _dcimSend('PUT', `/api/v1/dcim/catalog/${
            encodeURIComponent(_dcCatEditing.uid)}`, cuerpo)
        : await _dcimSend('POST', '/api/v1/dcim/catalog', cuerpo);
    if (!r.ok) return showToast(r.error || t('save_failed'), 'danger');
    hideInfoModal();
    showToast(t('saved'), 'success');
    _dcCatEditing = null;
    _dcCatCloneFrom = '';
    _dcCatOpen();
}
