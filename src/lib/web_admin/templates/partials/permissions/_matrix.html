{# ================================================================ Permissions — layout A: one grid, permissions × roles. What it is for: reading ACROSS roles. Built-in columns sit first in descending privilege, so a custom role is read against the yardstick it was cloned from and "editor plus audit_delete" is a shape you can see rather than reconstruct. State, filtering and saving all live in _state.html; this file only draws. ================================================================ #} /** One column header: role name, a lock for the read-only built-ins, and how many of * the catalog's permissions it grants. */ function _permMatrixHead(roles) { let h = `${esc(t('col_permission'))}`; for (const { uid, rd } of roles) { const ro = !_permCanEdit(uid); const lock = rd.builtin ? ` ` : ''; const off = (!rd.builtin && rd.enabled === false) ? ` ${esc(t('status_disabled'))}` : ''; const n = _permCounts(uid); const scoped = n.scoped ? ` ${n.scoped}` : ''; h += `
${esc(_permRoleName(uid))}${lock}${off}
${n.global}${scoped} `; } return h; } /** Group separator row. Each role's cell carries how many of the group's VISIBLE * permissions it holds; for an editable role that count is the button that grants or * revokes the whole group at once. */ function _permMatrixGroupRow(groupKey, perms, roles) { let h = ` ${esc(t(groupKey) || groupKey)} ${perms.length ? `${perms.length}` : ''}`; for (const { uid } of roles) { // The filter can leave a group with only its per-instance rows visible: no counter // and no bulk button then, rather than a "0/0" that toggles nothing. if (!perms.length) { h += ''; continue; } const held = perms.filter(p => _permHas(uid, p)).length; const label = `${held}/${perms.length}`; h += _permCanEdit(uid) ? `` : `${label}`; } return h + ''; } function _permMatrixPermRow(perm, roles) { // The second line is the permission's DESCRIPTION, as in the two-pane view. It used to // be the raw flag (`audit_delete`), which reads like a missing translation sitting // where the explanation should be. The flag is still there, on hover: it is what the // API speaks, so it stays reachable without taking the line that explains the thing. const hint = _permHint(perm); let h = ` ${esc(_permLabel(perm))} ${hint ? `
${esc(hint)}
` : ''}`; for (const { uid } of roles) { const ro = !_permCanEdit(uid); h += ` `; } return h + ''; } /** The per-instance overrides of a group: a caption row, then one row per (item, action) — * because in this layout a row IS a permission and the columns are already spent on roles. * The role modal can afford its items × actions table; it only ever shows one role. */ function _permMatrixResRows(groupKey, roles) { const { spec, rows } = _permVisibleResRows(groupKey); if (!spec || !rows.length) return ''; // Collapsed by default and folded from the caption row: N modules × 4 actions unfolded // would bury the catalog rows this layout exists to compare. The count stays visible so // a closed block still says how much is in there. const shut = _permResCollapsed(spec.prefix); let h = ` ${esc(spec.info())} ${rows.length}`; if (shut) return h; for (const row of rows) { const sub = row.sub?.text ? `${esc(row.sub.text)}` : ''; for (const action of _PERM_ACTIONS) { const key = `${spec.prefix}.${row.id}.${action}`; h += `${esc(row.name)} · ${esc(_permActionLabel(spec, action))}${sub}`; for (const { uid } of roles) { const ro = !_permCanEdit(uid); h += ` `; } h += ''; } } return h; } function _permMatrixHtml() { const roles = _permRoles(); let body = '', shown = 0; for (const [groupKey] of _permGroups()) { const perms = _permVisibleIn(groupKey); const res = _permMatrixResRows(groupKey, roles); if (!perms.length && !res) continue; // whole group filtered out shown += perms.length + (res ? 1 : 0); body += _permMatrixGroupRow(groupKey, perms, roles); for (const perm of perms) body += _permMatrixPermRow(perm, roles); body += res; } if (!shown) return _permNoMatchHtml(); return `
${_permMatrixHead(roles)}${body}
`; } /** Shared by both layouts: nothing survived the search / "only differences" filter. */ function _permNoMatchHtml() { return `
${esc(_permOnlyDiff() ? t('perm_no_differences') : t('perm_no_match'))}
`; }
