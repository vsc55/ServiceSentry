{# ================================================================ Users — access: what each account can actually do. The Users table has a role column, and the role column is not the answer. A user holds a role directly AND inherits every role their groups grant, so an account whose column says "viewer" and which sits in a group mapped to admin is an admin — and the table says viewer. Finding that out meant opening Groups, reading member lists and holding the composition in your head, which is exactly the work an access review is supposed to be. So: one row per user, the direct role, what the groups add, and a flag when the two are not the same. Admin reached through a group is called out on its own, because it is the only difference that changes what somebody can do to everything. Disabled groups grant nothing here, because they grant nothing on the server either. An access view that over-reports is useless in a different way than one that under-reports: it makes you chase access that does not exist, and hides the one that does behind noise. ================================================================ #} function _usrViewAccess(unames, ctx) { if (!unames.length) return ''; // Built-in identities are listed in the table and the cards — they are part of the system // and the audit log names them — but not here: this view answers "what can this account // do", and they hold nothing. Two permanent rows reading "None / — / —" would pad the // count chips of an access review with entries that can never be an answer. const rows = unames.filter(u => !usersData[u]?.builtin) .map(u => ({ u, eff: _accEffective(u) })); const viaGroups = rows.filter(r => r.eff.extra.length).length; const admins = rows.filter(r => r.eff.adminDirect || r.eff.adminViaGroup).length; const hiddenAdmins = rows.filter(r => !r.eff.adminDirect && r.eff.adminViaGroup).length; const chips = `${_summaryChip('bi-people', t('col_user'), rows.length)} ${admins ? _summaryChip('bi-shield-lock', t('usr_count_admins'), admins, hiddenAdmins ? 'text-bg-warning' : '') : ''} ${viaGroups ? _summaryChip('bi-diagram-3', t('usr_count_via_groups'), viaGroups) : ''} ${hiddenAdmins ? ` ${esc(tf('usr_admin_via_group_n', hiddenAdmins))}` : ''}`; return `
${_summaryHeader(chips)}
${rows.map(_usrAccessRow).join('')}
${esc(t('col_user'))} ${esc(t('usr_col_direct'))} ${esc(t('usr_col_via_groups'))} ${esc(t('col_groups'))}
`; } function _usrAccessRow(r) { const u = usersData[r.u] || {}; const eff = r.eff; const groups = _accGroupsOf(r.u); // A group that is switched off is shown — it IS how the account is configured — but // struck through the fact that it grants nothing today, so the row cannot be read as // access the user has. const groupCell = groups.length ? groups.map(g => ` ${esc(g.name || g.key)} ${g.enabled === false ? `` : ''}`).join(' ') : '—'; const viaCell = eff.via.size ? [...eff.via.entries()].map(([role, gs]) => ` ${_accRoleBadge(role)} ${esc(gs.join(', '))} `).join('') : '—'; return ` ${esc(r.u)} ${u.enabled === false ? `${esc(t('status_disabled'))}` : ''} ${(!eff.adminDirect && eff.adminViaGroup) ? `${esc(t('usr_admin_via_group'))}` : ''} ${_accRoleBadge(eff.direct)} ${viaCell} ${groupCell} `; }
