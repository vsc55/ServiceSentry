{# ================================================================ fail2ban › Banned IPs — by network. Forty bans are usually three networks. Whoever is knocking rotates the last octet, so the list shows forty separate incidents and the shape — "all of this is one /24" — only exists when they sit together. It is also the shape that changes the decision: you stop banning addresses one at a time and put the range on the list. A network with a single ban is not news, so the ones with several lead and the singles fall to the bottom. Nothing here bans anything: this is a reading surface, and the controls that ban live on the table beside it. ================================================================ #} function _ipbBansNetworks(bans) { if (!bans.length) return ''; const by = new Map(); for (const b of bans) { const net = _ipbNetwork(b.ip); let g = by.get(net); if (!g) { g = { net, ips: [], offenses: 0, levels: new Set() }; by.set(net, g); } g.ips.push(b); g.offenses += Number(b.offenses || 0); if (b.level != null && b.level !== '') g.levels.add(String(b.level)); } // Busiest network first: the row with eleven addresses under it is the reason to open // this view, and a network with one is the same information the table already gave. const rows = [...by.values()].sort((a, b) => (b.ips.length - a.ips.length) || a.net.localeCompare(b.net)); const clustered = rows.filter(r => r.ips.length > 1).length; const chips = `${_summaryChip('bi-slash-circle', t('ipban_manage_title'), bans.length)} ${_summaryChip('bi-diagram-2', t('ipb_count_networks'), rows.length)} ${clustered ? _summaryChip('bi-layers', t('ipb_count_clustered'), clustered, 'text-bg-warning') : ''}`; return `
${_summaryHeader(chips)}
${rows.map(_ipbNetworkRow).join('')}
${esc(t('ipb_col_network'))} ${esc(t('ipb_col_addresses'))} ${esc(t('ipban_col_offenses'))} ${esc(t('ipban_col_ip'))}
`; } function _ipbNetworkRow(g) { const many = g.ips.length > 1; // The addresses are listed, not counted: which ones is what you copy into a range, and // a network with thirty would otherwise be a paragraph. const addrs = _chipList(g.ips, b => _chipText(b.ip), 8); return ` ${esc(g.net)} ${g.ips.length} ${g.offenses || '—'} ${addrs} `; }
