{# ================================================================ Groups — access: what each group actually hands out, and to whom. A group is a machine with two ends: members in, roles out. The table shows both ends as counts and badges in separate columns, which is enough to read one group and not enough to see the three states that matter: * a group that grants roles and has NO members. It does nothing, and it is the one a new hire gets added to next month, inheriting whatever it was left holding. * a group with members and NO roles. Those people are in a bucket that gives them nothing, which is usually a half-finished migration rather than a decision. * a DISABLED group with both. It looks fully configured and grants nothing at all, because the server skips it — the difference between "configured" and "in effect", and the one that makes an access review wrong if it is missed. The admin-granting groups lead, because membership of one is the strongest thing on this page: adding a user there is a bigger act than editing their role, and it is done from a different screen. ================================================================ #} function _grpViewAccess(names) { if (!names.length) return ''; const rows = names.map(key => { const g = groupsData[key] || {}; const roles = g.roles || []; return { key, g, roles, members: (g.members || []).length, enabled: g.enabled !== false, admin: roles.some(_accIsAdminRole), }; }); // Admin-granting first, then the ones that hand out nothing or reach nobody, then the // rest by size: the order is "what would surprise you", not the alphabet. const rank = r => (r.admin ? 0 : (!r.roles.length || !r.members ? 1 : 2)); rows.sort((a, b) => rank(a) - rank(b) || b.members - a.members || String(a.g.name || a.key).localeCompare(String(b.g.name || b.key))); const admins = rows.filter(r => r.admin).length; const idle = rows.filter(r => !r.members || !r.roles.length).length; const off = rows.filter(r => !r.enabled).length; const chips = `${_summaryChip('bi-diagram-3', t('grp_count_groups'), rows.length)} ${admins ? _summaryChip('bi-shield-lock', t('grp_count_admin'), admins, 'text-bg-warning') : ''} ${idle ? _summaryChip('bi-slash-circle', t('grp_count_idle'), idle) : ''} ${off ? _summaryChip('bi-toggle-off', t('grp_count_disabled'), off) : ''}`; return `
${_summaryHeader(chips)}
${rows.map(_grpAccessRow).join('')}
${esc(t('col_group_name'))} ${esc(t('col_group_members'))} ${esc(t('col_group_roles'))} ${esc(t('grp_col_effect'))}
`; } function _grpAccessRow(r) { // What this group does TODAY, said in one cell: the three states the table's separate // columns can only be read as a pair of numbers. let effect; if (!r.enabled) effect = `${esc(t('grp_effect_disabled'))}`; else if (!r.roles.length) effect = `${esc(t('grp_effect_no_roles'))}`; else if (!r.members) effect = `${esc(t('grp_effect_no_members'))}`; else effect = `${esc(tf('grp_effect_grants', r.members))}`; return ` ${esc(r.g.name || r.key)} ${r.g.builtin ? `` : ''} ${r.admin ? `${esc(t('grp_grants_admin'))}` : ''} ${r.members ? `${r.members}` : '0'} ${r.roles.length ? r.roles.map(_accRoleBadge).join(' ') : '—'} ${effect} `; }
