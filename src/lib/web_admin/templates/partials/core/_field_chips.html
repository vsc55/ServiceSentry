{# ================================================================ Multi-value fields: removable chips, IP/CIDR checking, and who a notification goes to. One control with three jobs stacked on it — a comma-separated list drawn as chips, the validation for the ones that hold addresses (mirroring the backend's `ipaddress` check, which stays the authority), and the typeahead that turns a panel user or group into a chip with a name on it. tf('invalid_ip_value', bad.join(', ')) : 'Invalid IP: ' + bad.join(', ')), 'danger'); } } function chipsHtml(key, value) { return _multiList(value).map(v => { // Recipient tokens render as the group/user name + an icon (they expand to // email(s) on send). `group:` → members' emails, `user:` → that user's // email. cls = 'text-bg-info'; } else { inner = esc(v); } return ` ${inner}`; }).join(''); } // ── Recipient typeahead (panel users + groups) for `multi` fields with suggest ── let _recipSuggest = null; // {users:[{uid,email,name}], groups:[{uid,name}]} const _recipGroupMap = {}; // uid -> name (chip labels) const _recipUserMap = {}; // uid -> name (chip labels) function _recipGroupLabel(uid) { return _recipGroupMap[uid] || (t('notif_recipient_group_unknown') || uid); } function _recipUserLabel(uid) { return _recipUserMap[uid] || (t('notif_recipient_user_unknown') || uid); _multiSuggestClose(); const box = document.querySelector(`[data-multi-chips="${CSS.escape(key)}"]`); const inp = box && box.querySelector('input'); if (inp) { inp.value = ''; inp.focus(); } } function _multiSuggestRender(box, inp, users, groups) { _multiSuggestClose();
