{# ================================================================ Generic renderer for a module-contributed section (schema.json → __page__). A module that declares no `render` gets this: the core paints whatever its `page_data()` hook returned, so contributing a section costs NO front-end code. A module that wants a different look declares `render` and ships its own web/_ui.html instead — this file is never involved then. The shape it renders (everything module-supplied, nothing hardcoded here): {sections: [{id, name, state, counts:{ok,warn,error,total}, rows: [{name, state, message, metrics:{k: v}}]}], counts: {...}, live: bool, items: [{key, label}]} Live refresh: only when the page declares `refresh` (a watchful action name). The action must answer in the same shape, so there is one renderer, not two. ================================================================ #} const _MP_STATE_CLS = { ok: 'success', warn: 'warning', error: 'danger' }; const _mpState = {}; // {pageId: {spec, data, busy}} /** Section entry point for every module page without its own renderer. The wiring * passes the page descriptor, so one function serves all of them. */ async function renderModulePage(spec) { const st = _mpState[spec.id] || (_mpState[spec.id] = { spec, data: null, busy: false }); const c = document.getElementById(spec.id + '-container'); if (!c) return; if (!st.data) { c.innerHTML = `
${esc(t('loading') || 'Loading…')}
`; } const r = await apiGet('/api/v1/modules/page/' + encodeURIComponent(spec.module)); st.data = (r && r.data) || {}; _mpRender(spec.id); } /** Ask the module to fetch live data now (its declared `refresh` action). */ async function _mpRefresh(pageId, itemKey) { const st = _mpState[pageId]; if (!st || st.busy || !st.spec.refresh) return; st.busy = true; _mpRender(pageId); // repaint with the button disabled const items = (st.data && st.data.items) || []; const key = itemKey || (items[0] && items[0].key); const mod = st.spec.module; // The module config is keyed either "watchfuls." or "" depending on how it // was written — the backend accepts both, so look under both here too. const modCfg = modulesData?.[`watchfuls.${mod}`] || modulesData?.[mod] || {}; const cfg = { ...((modCfg.list || {})[key] || {}), _item_key: key, _lang: CURRENT_LANG }; const r = await apiPost( `/api/v1/modules/watchfuls/${encodeURIComponent(mod)}/${encodeURIComponent(st.spec.refresh)}`, cfg); const d = r && r.data; st.busy = false; if (d && d.ok) st.data = { ...st.data, sections: d.sections, counts: d.counts, live: true }; else showToast(`${esc(t('page_refresh_error'))}: ${esc((d && d.message) || '')}`, 'danger'); _mpRender(pageId); } /** A row's measurements, as the check published them (used %, days left, score…). * The core does not know what they mean — it shows the module's own key/value. */ function _mpMetrics(row) { const m = row.metrics || {}; const keys = Object.keys(m).filter(k => k !== 'alert').slice(0, 4); if (!keys.length) return ''; return `
${keys.map(k => `${esc(k)}: ${esc(String(m[k]))}` ).join('')}
`; } function _mpRender(pageId) { const st = _mpState[pageId]; const c = document.getElementById(pageId + '-container'); if (!st || !c) return; const d = st.data || {}; const sections = d.sections || []; const counts = d.counts || { ok: 0, warn: 0, error: 0, total: 0 }; const refreshBtn = st.spec.refresh ? ` ` : ''; const head = `
${esc(st.spec.label || pageId)} ${esc(d.live ? t('page_live') : t('page_cached'))}
${['ok', 'warn', 'error'].map(k => counts[k] ? `${counts[k]}` : '').join(' ')} ${refreshBtn}
`; if (!sections.length) { c.innerHTML = head + `
${esc(t('page_no_data'))}
`; return; } // One card per section the module returned — the core lays out, the module decides // what the sections are and what they are called. c.innerHTML = head + sections.map(s => `
${esc(s.name)}
${s.counts.ok}/${s.counts.total}
${(s.rows || []).map(row => `
${esc(row.name)}
${esc(row.message || '')}
${_mpMetrics(row)}
`).join('')}
`).join('') + ``; }
