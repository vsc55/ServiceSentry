{# ================================================================ Module pages — the layouts, and what they share. A module page is always the same shape (sections of rows, each row with a state, a message and the measurements the check published), so the layouts belong to the CORE renderer rather than to any one module. M365 and Azure get them from the same code, and a module contributing a page tomorrow gets them without writing front-end at all. What a view may decide is chrome and navigation. What a row MEANS — its state, its measurements, how it is drawn — stays in `_mpRow`, used verbatim by all four. A view that drew a row itself would be free to disagree with the view next to it about the same check. State is per PAGE, not global: wanting the table for Azure and the board for M365 at the same time is normal, and one shared setting would make each visit undo the other. ================================================================ #} const MP_VIEWS = [ { id: 'board', label_key: 'page_view_board', icon: 'bi-columns-gap', render: '_mpViewBoard' }, { id: 'split', label_key: 'page_view_split', icon: 'bi-layout-sidebar-inset', render: '_mpViewSplit' }, { id: 'table', label_key: 'page_view_table', icon: 'bi-list-ul', render: '_mpViewTable' }, { id: 'cards', label_key: 'page_view_cards', icon: 'bi-grid-3x2-gap', render: '_mpViewCards' }, ]; function _mpViewId(pageId) { const stored = localStorage.getItem('ss_page_view_' + pageId); return MP_VIEWS.some(v => v.id === stored) ? stored : MP_VIEWS[0].id; } /** Switching view redraws what is already on screen. A module page's data costs a Graph * call to refresh; making a way of LOOKING at it spend one would be charging the reader * for a decision about layout. */ function setModulePageView(pageId, id) { if (!MP_VIEWS.some(v => v.id === id)) return; localStorage.setItem('ss_page_view_' + pageId, id); _mpRender(pageId); } function _mpViewSwitcher(pageId) { const cur = _mpViewId(pageId); return `
${MP_VIEWS.map(v => ` `).join('')}
`; } // ── Filters, per page and shared by its views ──────────────────────────────── function _mpFilterTerm(pageId) { return _mpState[pageId]?.term || ''; } function _mpOnlyProblems(pageId) { return !!_mpState[pageId]?.onlyBad; } function _mpSetFilter(pageId, term) { const st = _mpState[pageId]; if (!st) return; st.term = String(term || ''); _mpRender(pageId); const el = document.getElementById('mpSearch_' + pageId); if (el) { el.focus(); el.value = st.term; el.setSelectionRange(el.value.length, el.value.length); } } function _mpToggleProblems(pageId, on) { const st = _mpState[pageId]; if (!st) return; st.onlyBad = !!on; _mpRender(pageId); } /** The rows of a section after the filters. Kept separate from the section's own counts: * the badge still says 4/8 while the list shows the four that are not OK, because hiding * the passing rows must not also hide that they exist. */ function _mpShownRows(pageId, s) { const q = _mpFilterTerm(pageId).trim().toLowerCase(); let rows = s.rows || []; if (_mpOnlyProblems(pageId)) rows = rows.filter(r => r.state !== 'ok'); if (q) rows = rows.filter(r => String(r.name || '').toLowerCase().includes(q) || String(r.message || '').toLowerCase().includes(q)); return rows; } /** Sections worth drawing, worst first. * * A section whose rows are all filtered away is dropped: an empty card headed "Licenses" * tells you nothing except that you filtered. The order is part of the answer — the reader * came to find what is wrong, not to scroll past what is not. */ function _mpVisibleSections(pageId, sections, sort = true) { const out = (sections || []) .map(s => [s, _mpShownRows(pageId, s)]) .filter(([s, rows]) => rows.length || (!_mpOnlyProblems(pageId) && !_mpFilterTerm(pageId).trim() && !(s.rows || []).length)); if (!sort) return out; const rank = s => (s.state === 'error' ? 0 : s.state === 'warn' ? 1 : 2); return out.sort((a, b) => rank(a[0]) - rank(b[0])); } // ── The one bar above the results ──────────────────────────────────────────── /** Totals on the left, the controls that change what you see of them on the right — the * same arrangement Status settled on, for the same reason: a filter that hides most of the * page is only honest while the count of everything sits beside the switch that hid it. */ function _mpResultsHeader(pageId, counts) { const bit = (n, cls, ico, label) => n ? `${n} ${esc(label)}` : ''; return `
${counts.total || 0} ${esc(t('page_checks'))} ${bit(counts.ok, 'text-success', 'bi-check-circle-fill', t('status_ok'))} ${bit(counts.warn, 'text-warning-emphasis', 'bi-exclamation-triangle-fill', t('status_warning'))} ${bit(counts.error, 'text-danger', 'bi-x-circle-fill', t('status_error'))}
${_mpViewSwitcher(pageId)}
`; } function _mpEmptyState(pageId, icon) { const filtering = _mpOnlyProblems(pageId) || !!_mpFilterTerm(pageId).trim(); return `

`; } // ── View: board ────────────────────────────────────────────────────────────── // One tile per section with its number, then only the rows that need attention, from every // section at once. ${esc(t('st_no_problems'))}
`}`; } /** A section as a single figure. For a section with many // rows — licences, several tenants — where the stacked cards make you scroll past the ones // you are not reading. function _mpViewSplit(pageId, st, c) { const shown = _mpVisibleSections(pageId, st.data.sections || []); if (!shown.length) { c.innerHTML = _mpEmptyState(pageId, st.spec.icon); return; } const ids = shown.map(([s]) => s.id); const sel = ids.includes(st.section) ? st.section : ids[0]; const [cur, rows] = shown.find(([s]) => s.id === sel); c.innerHTML = `
${esc(t('page_sections'))}
${shown.map(([s]) => ` `).join('')}
${esc(cur.name)} ${_mpGroupSelect(pageId, cur)} ${cur.counts.ok}/${cur.counts.total}
${rows.map(r => _mpRow(r, cur.chart)).join('')}
`; } function _mpSelectSection(pageId, sectionId) { const st = _mpState[pageId]; if (!st) return; st.section = sectionId; _mpRender(pageId); } // ── View: table ────────────────────────────────────────────────────────────── // One row per CHECK across every section. The stacked cards group by section, which is right // for reading a subject and wrong for triage: "what needs attention" is a question about // checks, and the answer was spread across however many cards happened to contain one. function _mpViewTable(pageId, st, c) { const shown = _mpVisibleSections(pageId, st.data.sections || []); const rank = { error: 0, warn: 1, ok: 2 }; const flat = []; for (const [s, rows] of shown) for (const r of rows) flat.push([s, r]); flat.sort((a, b) => (rank[a[1].state] ?? 3) - (rank[b[1].state] ?? 3) || String(a[1].name).localeCompare(String(b[1].name))); if (!flat.length) { c.innerHTML = _mpEmptyState(pageId, st.spec.icon); return; } c.innerHTML = `
${flat.map(([s, r]) => ` `).join('')}
${esc(t('mod_state'))} ${esc(t('page_section'))} ${esc(t('st_check'))}

${r.message ? `
${esc(r.message)}
` : ''} 		${_mpMetrics(r)}${_mpRowChart(s.chart, r)}
`; }
