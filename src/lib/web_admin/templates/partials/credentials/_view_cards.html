{# ================================================================ Credentials — cards: one card per credential. The same card the rest of the panel uses for an entity (users, roles, groups), so a credential looks like a thing rather than like a row. It costs the metadata columns and buys the two facts you actually chase a credential by — its type and the identity inside it (the SSH user, the app id) — at a glance, plus room for the description the table can only show truncated. The catalogue is small by nature: a handful of identities reused everywhere. This is the view for reading it, and the table is the one for auditing it. ================================================================ #} function _credViewCards(uids, ctx) { return _cardGrid(uids.map(uid => { const cr = credentialsData[uid] || {}; const ident = _credIdentityText(cr); // Chips: the identity the secret belongs to, and when it last changed. A secret's own // value is never shown anywhere — the API masks it before it leaves the server. const chips = [ ident ? ` ${esc(ident)}` : '', cr.updated_at ? ` ${esc(_fmtDateTime(cr.updated_at))}` : '', ].join(''); return _entityCard({ icon: 'bi-key', accentClass: 'cyan', selected: _selectedCredentials.has(uid), toggle: ctx.canDelete ? `_credentialsToggleOne(${jsStr(uid)}, this.checked)` : '', title: `${esc(cr.name || uid)}`, badges: `${_credTypeBadge(cr.ctype)}${_credDisabledBadge(cr)}`, subtitle: cr.description ? esc(cr.description) : '', chips, actions: ctx.showActions ? _credActionsHtml(uid, ctx) : '', dimmed: cr.enabled === false, }); }).join('')); }
