{# ================================================================ Credentials — by type: the catalogue split by what KIND of secret each one is. The table shows an SSH identity and a tenant app registration as two rows of the same shape, and they are not the same animal: one is a user + key you reach a machine with, the other is an application with consented permissions and no host behind it at all. They are created differently, they expire differently, and they are used by different halves of the panel. Sorting by the Type column interleaves them into a single list; grouping puts each kind in its own block with its own count. The type is also where the catalogue's shape shows: types come from the modules themselves (`__credential__` in schema.json), so this view answers "which kinds of secret does this installation even know about" — including the ones nothing has filled in yet, listed at the bottom. THAT line is computed over the whole catalogue rather than the page on screen, and says so, because "no credentials of this type" is a statement about the installation and would be a lie if paging changed it. ================================================================ #} function _credViewTypes(uids, ctx) { // A summary body: the factory hands it every row the filters left standing, not the page // (see the 'summary' mode in createListTable). Grouping a page would make each group's // count a statement about the pagination — "3 SSH credentials" when there are ten, and a // different three on the next page. const groups = new Map(); for (const uid of uids) { const ct = (credentialsData[uid] || {}).ctype || 'ssh'; if (!groups.has(ct)) groups.set(ct, []); groups.get(ct).push(uid); } // Registry order first (the order the backend publishes the types in, ssh leading), then // any type present in the data but no longer declared — a module was removed and its // credentials outlived it, which is exactly the case worth seeing rather than hiding. const known = _credTypeKeys(); const order = known.filter(ct => groups.has(ct)) .concat([...groups.keys()].filter(ct => !known.includes(ct))); const body = order.map(ct => { const rows = groups.get(ct); const spec = _CRED_TYPES[ct] || {}; const orphanType = !known.includes(ct); const modChip = (spec.module && spec.module !== '__core__') ? ` ${esc(modulePrettyName(spec.module, {}))}` : ''; const gone = orphanType ? ` ${esc(t('cred_type_unknown'))}` : ''; return `
${_credTypeIcon(ct)} ${esc(_credTypeLabel(ct))} ${rows.length} ${modChip}${gone}
${rows.map(uid => _credTypeRow(uid, ctx)).join('')}
`; }).join(''); return `
${body}${_credUnusedTypesNote()}
`; } function _credTypeRow(uid, ctx) { const cr = credentialsData[uid] || {}; const ident = _credIdentityText(cr); return `
${esc(cr.name || uid)} ${_credDisabledBadge(cr)} ${ident ? `${esc(ident)}` : ''} ${esc(cr.description || '')} ${ctx.showActions ? _credActionsHtml(uid, ctx) : ''}
`; } /** The types this installation declares and has no credential for — over the WHOLE * catalogue, not the page. Nothing is wrong with an empty type; it is the answer to "what * else could I define here", which is otherwise only visible inside the New-credential * form's type dropdown. */ function _credUnusedTypesNote() { const filled = new Set(Object.values(credentialsData).map(c => c.ctype || 'ssh')); const empty = _credTypeKeys().filter(ct => !filled.has(ct)); if (!empty.length) return ''; return `
${esc(t('cred_types_empty'))} ${empty.map(ct => `${esc(_credTypeLabel(ct))}`).join('')}
`; }
