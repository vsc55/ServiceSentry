{# ================================================================ Credentials — usage: who still references each credential. The question the catalogue cannot answer about itself, and the one that decides whether a secret is safe to change or safe to delete. A credential's consumers are not part of the credential: they live in the hosts store (a host's SSH profile) and inside every module's config (a check's inline cred_uid). Until now the only way to see them was to open a credential and click its Usage tab — one credential at a time, which is fine for "can I delete this one" and useless for "what is this catalogue full of". Seen as a whole it answers the question that rots a credential store: a secret nobody references is a secret nobody rotates, and it stays valid. Those rows are marked, and the banner counts them over the WHOLE catalogue rather than the page on screen — an orphan count that shrank as you paged would be worse than no count at all. The rows keep the order the table header decides. Floating the unused ones to the top would read well and silently override the sort the user chose two views ago; the badge says the same thing without moving anything. ================================================================ #} function _credViewUsage(uids, ctx) { // The usage map is a separate fetch: ask once, draw a waiting state until it lands. if (!_credEnsureUsage()) return _credUsagePending(); const orphans = _credOrphanUids().length; const total = Object.keys(credentialsData).length; const banner = orphans ? `
${esc(tf('cred_unused_n', orphans, total))}
` : `
${esc(t('cred_all_used'))}
`; return `
${banner}
${uids.map(uid => _credUsageRow(uid, ctx)).join('')}
${esc(t('cred_name'))} ${esc(t('cred_type'))} ${esc(t('cred_usage_hosts'))} ${esc(t('cred_usage_checks'))} ${esc(t('col_actions'))}
`; } function _credUsageRow(uid, ctx) { const cr = credentialsData[uid] || {}; const use = _credUsageFor(uid); const unused = !use.hosts.length && !use.checks.length; return ` ${esc(cr.name || uid)} ${_credDisabledBadge(cr)} ${unused ? `${esc(t('cred_unused'))}` : ''} ${_credTypeBadge(cr.ctype)} ${_credConsumerChips(use.hosts.map(h => ({ text: h.name || h.uid })), 'bi-hdd')} ${_credConsumerChips(use.checks.map(c => ({ text: `${c.module} — ${c.label}` })), 'bi-list-check')} ${ctx.showActions ? _credActionsHtml(uid, ctx) : ''} `; } /** A consumer list as chips, cut to the first few with a "+n" tail: a credential wired into * forty checks would otherwise turn its row into a paragraph. The full list is still one * click away in the credential's own Usage tab. */ function _credConsumerChips(items, icon) { if (!items.length) return '—'; const shown = items.slice(0, 4).map(i => ` ${esc(i.text)}`).join(' '); const more = items.length > 4 ? `+${items.length - 4}` : ''; return `${shown}${more}`; }
