{# ================================================================ Backup — what the buttons on a row do: delete, verify, lock. Operations on ONE archive, and not part of drawing the list. Their own file because the render shell has a size limit that exists to stop a section growing sub-sections inside itself — restoring and the details dialog left for the same reason. Every one of them says what will happen rather than asking "are you sure?": that is not a question anybody can answer, and the one worth answering is what happens if I say yes. ================================================================ #} function deleteBackup(name) { showConfirmModal(t('backup_delete_confirm').replace('{}', name), async () => { const res = await apiDelete(`/api/v1/backups/${encodeURIComponent(name)}`); if (_apiOk(res)) { showToast(t('backup_deleted_ok')); renderBackups(); } else showToast(_apiErr(res), 'danger'); }); } /** Protect a copy from being deleted, or stop protecting it. * * Retention answers "how much history" and its two floors answer "never leave the task with * nothing". Neither can say *this particular archive* — which is what somebody means about the * copy taken before a migration, or the last one known to be good. * * Locking needs no confirmation: it takes nothing away, and it is undone by the same button. * UNLOCKING does, because what it does is hand the copy back to the counter that was about to * delete it — and the dialog says that instead of asking whether you are sure. */ function toggleBackupLock(name, locked) { const go = async () => { const res = await apiPost(`/api/v1/backups/${encodeURIComponent(name)}/lock`, {locked: !!locked}); if (_apiOk(res)) { showToast(t(locked ? 'backup_lock_ok' : 'backup_unlock_ok')); renderBackups(); } else showToast(_apiErr(res), 'danger'); }; if (locked) { go(); return; } showConfirmModal(t('backup_unlock_confirm').replace('{}', name), go); } /** Check a copy against its own checksums, and say which question failed. * "The file is intact" and "the contents are unchanged" are different failures with different * causes — a truncated transfer versus altered data — so the answer names which. */ async function verifyBackup(name) { const res = await apiPost(`/api/v1/backups/${encodeURIComponent(name)}/verify`, {}); if (!_apiOk(res)) { showToast(_apiErr(res), 'danger'); return; } const v = res.data || {}; if (v.ok) { showToast(t('backup_verify_ok').replace('{}', v.members)); return; } const why = v.file === 'bad' ? t('backup_verify_file_bad') : t('backup_verify_members_bad').replace('{}', (v.bad || []).length); showToast(why, 'danger'); } //: Lo que está marcado para borrar. Un `Set` de nombres y no un campo en cada fila: la lista se //: repinta sola mientras corre una copia, y un estado que viva en el DOM se pierde en cada //: repintado — que es justo mientras alguien está eligiendo. const _bkSel = new Set(); /** Repintar la lista y nada más. El mismo camino que usan el orden y el cambio de vista: la * lista no se vuelve a pedir al servidor por marcar una casilla. */ function _bkRepaint() { const c = document.getElementById('backup-container'); if (c) _renderBackupPane(c); } function bkSelectOne(name, on) { if (on) _bkSel.add(name); else _bkSel.delete(name); _bkRepaint(); } /** Todas, o ninguna — y «todas» son **las que se pueden borrar**. * * Una copia con candado no entra: la casilla de su fila está desactivada, y una cabecera que * la metiera en el montón llevaría a un diálogo que promete borrar cinco y borra cuatro. */ function bkSelectAll(on) { const nombres = _backupsFor(_backupView).filter(b => !b.locked).map(b => b.name); if (on) nombres.forEach(n => _bkSel.add(n)); else _bkSel.clear(); _bkRepaint(); } /** La barra que aparece al elegir la primera. No ocupa sitio mientras no hay nada elegido. */ function _bkSelBarHtml() { const n = _bkSel.size; if (!n) return ''; return `
${esc(n === 1 ? t('backup_sel_one') : tf('backup_sel_n', n))}
`; } /** Borrar las elegidas. * * Una petición por copia y no una en bloque, a propósito: cada archivo deja su propia línea en * la auditoría —que es la pregunta que se le hace a ese registro: «¿quién se llevó ésta?»— y * una que falle no se lleva a las demás por delante. * * Con una sola elegida se usa el diálogo de siempre, que **dice su nombre**: es más útil que * un «se eliminarán 1 copias», y es el mismo texto que sale por el botón de la fila. */ function bkDeleteSelected() { const nombres = [..._bkSel]; if (!nombres.length) return; if (nombres.length === 1) { deleteBackup(nombres[0]); _bkSel.clear(); return; } showConfirmModal(tf('backup_delete_many', nombres.length), async () => { let bien = 0; for (const name of nombres) { const res = await apiDelete(`/api/v1/backups/${encodeURIComponent(name)}`); if (_apiOk(res)) { bien += 1; _bkSel.delete(name); } } const mal = nombres.length - bien; // Un recuento y no un aviso por copia: veinte tostadas seguidas no las lee nadie, y lo // que hace falta saber es si faltó alguna. showToast(mal ? tf('backup_deleted_some', bien, mal) : tf('backup_deleted_n', bien), mal ? 'warning' : 'success'); renderBackups(); }); }
