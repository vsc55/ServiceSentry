{# ================================================================ Backup — the folder picker behind the `backup_dir` setting. Its own partial because the render shell has a size limit and this is a screen in its own right: a two-pane file browser, the "new folder" input, and the placeholder the empty setting shows. None of it is about the list of copies. It hangs off FIELD_PICKERS, a registry keyed by config path: the field renderer draws two hundred fields and must not know that one of them is a folder. ================================================================ #} // ── The folder picker ──────────────────────────────────────────────────────── // Registered against ONE config path rather than built into the field renderer: that file // draws two hundred fields and must not know that one of them is a folder. FIELD_PICKERS['web_admin|backup_dir'] = { icon: 'bi-folder2-open', title: 'backup_browse', open: (pathStr, current) => _openDirPicker(pathStr, current), // Filled in below. Empty until then rather than a guess: showing `/backups` as // literal text would put a path on screen that may not be the one in use. placeholder: '', }; /** Ask the server where copies go, and let the empty box say so. * * The setting's default is "" meaning `/backups`, and a blank box with a hint below * it leaves the operator to work out what that resolves to on THIS install. The server knows; * this asks once and repaints. */ async function _backupDirPlaceholder() { const fp = FIELD_PICKERS['web_admin|backup_dir']; if (!fp || fp.placeholder) return; const res = await apiGet('/api/v1/backups/browse'); if (!res || !res.configured) return; fp.placeholder = res.configured; // Only if the section is on screen: repainting Configuration from under an admin who is // typing into it would throw the edit away to add a placeholder. const pane = document.getElementById('tab-config'); if (pane && pane.classList.contains('active') && typeof renderConfig === 'function' && !_isDirty('config')) { renderConfig(); } } let _dirPickPath = ''; let _dirPickField = ''; async function _openDirPicker(fieldPath, current) { _dirPickField = fieldPath; // The modal opens FIRST, empty, and fills when the answer arrives. Awaiting the fetch // before showing it meant the button did nothing at all for as long as the server took — // and a button that does nothing is a button somebody presses again. _openBackupModal(t('backup_browse'), _dirPickBusy(current), t('backup_browse_use'), 'primary', () => {}); await _dirPickLoad(current || ''); } /** Draw the folder at *path* into the section's modal. * Directories only — the picker's job is to choose a folder, and every file name it showed * on the way would be information the screen did not need to do it. */ /** What the modal shows while the server is listing a folder. * A spinner alone says "something is happening"; naming the folder says WHAT, which is the * difference between waiting and wondering whether the click registered. */ function _dirPickBusy(path) { return `
${esc(t('backup_browse_scanning'))}
${path ? `${esc(path)}` : ''}
`; } async function _dirPickLoad(path) { // Every step down the tree shows it too, not only the first: walking into a folder on a // slow mount is exactly when the screen must not look frozen. const pane0 = document.getElementById('dirPickMain'); const busy = pane0 || document.getElementById('backupModalBody'); if (busy) busy.innerHTML = _dirPickBusy(path); const res = await apiGet('/api/v1/backups/browse?path=' + encodeURIComponent(path || '')); if (!res) { showToast(t('save_error'), 'danger'); return; } _dirPickPath = res.path || ''; // ── The left rail: where you can jump to ──────────────────────────────── // A rail and not a strip of buttons across the top: the roots are NAVIGATION, the same // job the sidebar does for the panel, and a row of them above the list competed with the // path for the one line that says where you are. const jump = (path, icon, label) => ``; // The configured folder first, and set apart: it is the one place the operator is most // likely to want back after wandering, and hunting for it among the drive letters is the // walk they just did in reverse. const cfgJump = res.configured ? `
${esc(t('backup_browse_current'))}
${jump(res.configured, 'bi-archive', _dirPickLeaf(res.configured))}
` : ''; const rail = `
${cfgJump}
${esc(t('backup_browse_places'))}
${(res.roots || []).map(r => jump(r, 'bi-hdd', r)).join('')}
`; const up = res.parent ? `` : ''; const rows = (res.dirs || []).map(d => ``).join('') || `
${esc(t('backup_browse_empty'))}
`; // Writability is stated EITHER WAY, not only when it is missing. "No warning" and "I did // not check" look identical on screen, and this is the one property that decides whether // the folder can hold a backup at all. const note = !res.readable ? `${esc(t('backup_browse_denied'))}` : (res.path ? (res.writable ? `${esc(t('backup_browse_rw'))}` : `${esc(t('backup_browse_ro'))}`) : ''); // Creating one is offered only where it can succeed: an input that always fails is worse // than no input, because it looks like the folder is the problem. const mk = res.readable && res.writable && res.path ? `
` : ''; // One row for "where am I" and "can I write here", fixed height. Stacked, the badge came // and went with the folder and the whole pane moved a line up and down with it. const main = `
${up} ${esc(_dirPickPath || '—')} ${note}
${mk}
${rows}
`; // Already open? Then only the RIGHT pane changes. Rebuilding the whole body made the // modal collapse to nothing and grow back on every click, because Bootstrap re-centres a // dialog whose content it has just been handed — and the rail, which did not change, was // being thrown away and rebuilt to show the same six drives. const pane = document.getElementById('dirPickMain'); if (pane) { pane.innerHTML = main; _dirPickMarkRail(); return; } _dirPickPendingMark = true; _openBackupModal(t('backup_browse'), `
${rail}
${main}
`, t('backup_browse_use'), 'primary', () => { // Written through `updateField`, the same call the box itself makes, so the value // lands in `configData` and the form goes dirty — a picker that only painted the // input would be a choice that Save never sees. updateField('config', _dirPickField, _dirPickPath); renderConfig(); }, 'lg'); if (_dirPickPendingMark) { _dirPickPendingMark = false; _dirPickMarkRail(); } } /** Light the rail entry the current folder is INSIDE, not the one it equals. * * Exact matching lit `C:\` while you were at `C:\`, and went dark the moment you opened a * folder in it — losing the one thing the rail is for, which is saying where in the machine * you are. * * The DEEPEST match wins, so standing in `/mnt/nas/backups` lights `/mnt` and not `/`, and * the configured folder beats the drive it lives on. Compared on a separator boundary: a path * is not inside `/var` because it starts with those four letters (`/variable`). */ function _dirPickMarkRail() { const cur = String(_dirPickPath || ''); const items = Array.from(document.querySelectorAll('#dirPickRail .ss-rail-item')); let best = null; let bestLen = -1; for (const el of items) { const p = el.getAttribute('title') || ''; const stem = p.replace(/[\\/]+$/, ''); const inside = cur === p || cur === stem || cur.startsWith(stem + '\\') || cur.startsWith(stem + '/') || stem === ''; // "/" contains everything if (inside && stem.length > bestLen) { best = el; bestLen = stem.length; } } items.forEach(el => el.classList.toggle('active', el === best)); } let _dirPickPendingMark = false; /** The last component of a path, for a rail label. A full path in an 11rem column is an * ellipsis with nothing readable left in front of it; the tooltip carries the whole thing. */ function _dirPickLeaf(path) { // Both separators: a Windows path has no forward slash in it, so a class holding // only `/` split nothing and the rail showed the whole path where a name belongs. const parts = String(path || '').replace(/[\/]+$/, '').split(/[\/]/); return parts[parts.length - 1] || path; } /** Create a folder inside the one being shown, then walk into it. * Walking in is the point: the operator made it to put copies there, so leaving them looking * at the parent would be one more click to do what they already said. */ async function _dirPickMake() { const name = (document.getElementById('dirPickNew')?.value || '').trim(); if (!name) return; const res = await apiPost('/api/v1/backups/mkdir', {parent: _dirPickPath, name}); if (_apiOk(res)) { await _dirPickLoad((res.data || {}).path || _dirPickPath); } else showToast(_apiErr(res), 'danger'); }
