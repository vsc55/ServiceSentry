{# ================================================================ Backup — putting a copy back, and saying what that means first. Its own partial because the render shell has a size limit, and because this is one story from end to end: what will be overwritten, which build the copy came from, what to apply, and — afterwards — what the live schema could not take from it. Nothing here refuses a copy over its version. The schema moves on almost every build, and a panel that turned down "old" copies would be useless on the one day it is needed. What it does is make the jump visible before the button is pressed, and the losses visible after. ================================================================ #} /** Which way the version jumps, said before the button is pressed. * * Nothing is refused over it — the schema moves on almost every build, and a panel that * turned down "old" copies would be useless on the one day it is needed. But a copy from a * LATER build loses the columns this schema does not have yet, quietly, and that has to be a * decision rather than something discovered afterwards. */ function _bkVersionNote(b) { if (b.version_rel === 'newer') { return `
${esc(t('backup_restore_newer').replace('{}', b.app_version || '?') .replace('{}', _bkVersion || '?'))}
`; } if (b.version_rel === 'older') { return `
${esc(t('backup_restore_older').replace('{}', b.app_version || '?') .replace('{}', _bkVersion || '?'))}
`; } return ''; } /** Is this part FILES rather than tables? * * `config_file` and anything a module declared are files, so the table picker has no say in * them: a restore that chose no table at all is still a restore if one of those is ticked. */ function _bkIsFilePart(pid) { const meta = _backupParts.find(x => x.id === pid); return !meta || meta.kind !== 'db'; } async function openRestoreModal(name) { const b = _backups.find(x => x.name === name); if (!b) return; const avail = _backupParts.filter(p => (b.parts || []).includes(p.id)); // Asked for BEFORE the dialog opens, not when the fold is disclosed: the archive is on the // same disk the list came from, and a fold that spends half a second empty after being // clicked reads as a fold with nothing in it. const inside = await _bkLoadTables(name); _openBackupModal(t('backup_restore'), `

${esc(t('backup_restore_warn').replace('{}', name))}

${_bkVersionNote(b)}
${esc(t('backup_restore_parts'))}
${_backupPartChecks('bkRes', avail.map(p => ({...p, required: false})), b.parts)}
${_bkTablePickHtml(inside)} ${b.secrets ? '' : `
${esc(t('backup_restore_nosecrets'))}
`}`, t('backup_restore'), 'warning', // Wide, and the BODY scrolls: with the fold open this form is taller than the screen // on any install with a few dozen tables, and a dialog that clips instead of scrolling // puts the last group of them underneath its own buttons. () => _restoreBackup(name, _checkedParts('bkRes', avail), _bkChosenTables()), 'scroll'); // Wired after the body is in the DOM, and once: the part checkboxes are drawn by the shared // `_backupPartChecks`, which knows nothing about a fold under it — putting a handler in // there would hang it on the create form and the task form as well. avail.forEach(p => { const el = document.getElementById(`bkRes_${p.id}`); if (el) el.addEventListener('change', _bkTablePickChanged); }); _bkTablePickChanged(); // draw the counters before anybody touches anything } /** Start the restore and watch it, through the same job the copies use. * * It used to be awaited: on an install whose tables run to six figures of rows, that is a * screen saying nothing at all while every one of them is being replaced — the moment where * silence is most alarming, because what it is silent about is the install being overwritten. */ /** Would this restore put anything back at all? * * Nothing ticked is not a restore, and sending it would empty every table of the parts that * ARE ticked with nothing to refill them — the one outcome worse than not restoring. Refused * in the form rather than by the API, which cannot tell "the operator cleared the boxes" from * "a script means exactly this". A file part with no tables under it still counts: config.json * and a module's files come back without a single row being touched. */ function _bkNothingToRestore(parts, tables) { if (!parts.length) return true; return Array.isArray(tables) && !tables.length && !parts.some(p => _bkIsFilePart(p)); } async function _restoreBackup(name, parts, tables) { if (_bkNothingToRestore(parts, tables)) { showToast(t('backup_restore_nothing'), 'warning'); return; } // `tables` only travels when something was actually left out: with the fold untouched this // is the request it always was. const body = {parts}; if (Array.isArray(tables)) body.tables = tables; const res = await apiPost(`/api/v1/backups/${encodeURIComponent(name)}/restore`, body); const jobId = _apiOk(res) && (res.data || {}).job_id; if (!jobId) { showToast(_apiErr(res), 'danger'); return; } // No row in the list: nothing is being ADDED to it. Only the dialog, which is where the // form that started this already was. _bkStartWatching({taskName: name, manual: true, restore: true}, jobId); } /** What a finished restore has to say: how much went back, and what did not. * * A toast is the wrong shape for it — it says a number and disappears, and this is the one * thing somebody has to read. The page reloads when the dialog is CLOSED, not before: * everything on screen was read before the tables changed under it, so a reload is the honest * answer, but reloading out from under the report would make it pointless. */ function _bkRestoreDoneBody(job) { const rows = Object.values(job.tables || {}) .reduce((n, v) => n + (Number(v) || 0), 0); const skipped = Object.entries(job.skipped || {}); const lead = job.error ? job.error : skipped.length ? t('backup_restored_partial').replace('{}', rows) : t('backup_restored_ok').replace('{}', rows); const tone = job.error ? 'danger' : skipped.length ? 'warning' : 'success'; const list = skipped.map(([table, what]) => `
${esc(table)} ${esc(what.missing ? t('backup_skipped_table').replace('{}', what.rows) : t('backup_skipped_cols').replace('{}', (what.columns || []).join(', ')))}
`).join(''); return `
${esc(job.task || '')}
${esc(lead)}
${job.partial && !job.error ? `
${esc(t('backup_restored_tables_only'))}
` : ''} ${_bkStepsHtml(job.steps)} ${list} ${job.error ? '' : `
${esc(t('backup_restored_reload'))}
`}
`; }
