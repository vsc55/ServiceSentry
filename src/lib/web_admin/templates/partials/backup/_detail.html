{# ================================================================ Backup — everything one copy knows about itself. The row has five columns and they answer "which copy is this". This answers "what is in it, and can I trust it": every part with its outcome and row count, who made it and with what version, and the sha256 of each member. Read entirely from the MANIFEST the copy carries. Nothing here is computed by the screen — the copy is what somebody restores from months later, and a verdict worked out at display time is a verdict that did not travel with the file. ================================================================ #} function _bkRow(label, value, mono) { if (value === '' || value === null || value === undefined) return ''; return '' + esc(label) + '' + '' + esc(String(value)) + ''; } /** The parts, with what happened to each. The same checklist the running dialog shows, from * the manifest instead of from a job — a copy has to answer this long after the run that * made it is gone. */ function _bkStepsTable(steps) { if (!steps || !steps.length) return ''; return '
' + steps.map(st => { const meta = _backupParts.find(x => x.id === st.part); const icon = st.ok ? '' : ''; const detail = st.error ? '' + esc(st.error) + '' : '' + esc(t('backup_n_rows').replace('{}', st.rows)) + ''; return '
' + icon + '' + esc(_bkPartLabel(meta, st.part)) + '' + detail + '
'; }).join('') + '
'; } /** The tables it holds, biggest first: on a copy that came out smaller than expected, which * table is missing its rows is the first thing worth seeing. */ function _bkTablesHtml(tables) { const rows = Object.entries(tables || {}).sort((a, b) => b[1] - a[1]); if (!rows.length) return ''; return '
' + esc(t('backup_detail_tables').replace('{}', rows.length)) + '' + '' + rows.map(function (e) { return '' + ''; }).join('') + '
' + esc(e[0]) + '' + esc(String(e[1])) + '
'; } /** The per-member digests, behind a fold: a wall of hex nobody reads top to bottom. It exists * so `Verify` has something to compare against, not so anybody memorises it. */ function _bkDigestsHtml(sha) { const rows = Object.entries(sha || {}); if (!rows.length) { return '
' + esc(t('backup_detail_nosha')) + '
'; } return '
' + esc(t('backup_detail_sha').replace('{}', rows.length)) + '' + '' + rows.map(function (e) { return '' + ''; }).join('') + '
' + esc(e[0]) + '' + esc(e[1]) + '
'; } function openBackupDetail(name) { const b = _backups.find(x => x.name === name); if (!b) return; const when = b.mtime ? new Date(b.mtime * 1000).toLocaleString() : ''; const rows = Object.values(b.tables || {}).reduce((n, v) => n + (Number(v) || 0), 0); // The same four answers the list gives, including "not recorded" for a copy taken before // verdicts existed: the row and this dialog disagreeing about whether a copy is good is // worse than either of them being unsure. const statusLabel = b.status === 'partial' ? t('backup_status_partial') : b.status === 'error' ? t('backup_status_error') : b.status === 'ok' ? t('backup_status_ok') : t('backup_status_unknown'); const html = '' + _bkRow(t('backup_col_name'), b.name, true) + _bkRow(t('col_time'), when) + _bkRow(t('backup_col_size'), b.size_h) + _bkRow(t('backup_detail_status'), statusLabel) + _bkRow(t('backup_detail_rows'), rows) + _bkRow(t('col_user'), b.created_by) + _bkRow(t('backup_detail_version'), b.app_version, true) + _bkRow(t('backup_detail_engine'), b.engine) + _bkRow(t('backup_secrets'), b.secrets ? t('backup_secrets_in') : t('backup_secrets_out')) + '
' + _bkStepsTable(b.steps) + _bkTablesHtml(b.tables) + _bkDigestsHtml(b.sha256); _openBackupModal(b.name, html, t('close'), 'secondary', () => {}, 'lg'); }
