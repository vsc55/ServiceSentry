{# ================================================================ Audit — timeline: the log read as a log. The table gives every entry the same five columns, which is right when you are comparing entries and wrong when you are following them: reading a sequence of events in a grid means re-reading the same column headers on every line to find out what happened. Here the day is stated once, the clock runs down the left, and each entry is one sentence — who did what, from where. It is the same rows the table would draw, in the same order, on the same page: the sort control still decides the direction, and the day headers follow it (oldest-first reads forwards, newest-first reads backwards). Nothing here re-sorts anything, so switching view never silently changes which entries you are looking at. ================================================================ #} function _auditViewTimeline(entries, ctx) { if (!entries.length) return ''; // Day headings come from the ORDER the entries arrive in, never from re-sorting them. // Grouping by day and then sorting the days would quietly override the direction the // user picked in the sort control, and the page would stop matching the pagination band. let out = ''; let day = null; for (const e of entries) { const k = _auditDayKey(e.ts); if (k !== day) { day = k; out += `
${esc(_auditDayLabel(e.ts))}
`; } out += _auditTimelineRow(e, ctx); } return `
${out}
`; } function _auditTimelineRow(e, ctx) { const del = _auditDeleteBtn(e, ctx); // The rail: a hairline with a dot per entry, so a burst of activity reads as a cluster // rather than as a wall of identical rows. return `
${esc(_auditClock(e.ts))} ${_auditEventBadge(e.event)} ${esc(_auditActor(e))} ${e.ip ? `${esc(e.ip)}` : ''} ${auditSummaryHtml(e._idx, e.detail)} ${del ? `${del}` : ''}
`; }
