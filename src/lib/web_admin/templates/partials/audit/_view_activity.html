{# ================================================================ Audit — activity: when the panel is being used. A list sorted by time can show you a 03:00 login and an 11:00 login and they read identically — the timestamp is a value in a column, and nothing about it says "this is an odd hour". On a day × hour grid the same two entries are in different places, and the answer to "does anything happen here outside working hours" is a shape you see from across the room instead of a query you have to think of. It is a SUMMARY: every entry the filters left standing, not the page, and no pagination — page 2 of a heat map is not a thing. The colour carries a count and nothing else (no state, no severity), so it is one hue getting darker, never a red-to-green ramp: this view has no opinion about whether activity is good. Every cell states its exact number on hover, because a shade is a comparison and not a value. ================================================================ #} // A long-running panel accumulates years of log, and 700 rows of grid is not a view, it is a // scroll. The cap is stated on screen when it bites — a silent truncation would read as "this // is all there is". const _AUD_ACT_MAX_DAYS = 62; function _auditViewActivity(entries) { if (!entries.length) return ''; const days = new Map(); // dayKey → { label, hours: number[24], total } for (const e of entries) { const k = _auditDayKey(e.ts); let d = days.get(k); if (!d) { d = { key: k, label: _auditDayLabel(e.ts), hours: new Array(24).fill(0), total: 0 }; days.set(k, d); } d.hours[new Date(e.ts).getHours()]++; d.total++; } // Oldest at the top: a calendar reads downwards, whichever way the list views are sorted. // This is the view's own axis rather than the user's sort — the sort control is hidden // while it is on screen for exactly that reason. const all = [...days.values()].sort((a, b) => a.key.localeCompare(b.key)); const shown = all.slice(-_AUD_ACT_MAX_DAYS); const dropped = all.length - shown.length; let max = 0, peak = null; for (const d of shown) { for (let h = 0; h < 24; h++) { if (d.hours[h] > max) { max = d.hours[h]; peak = { d, h }; } } } const chips = ` ${esc(t('audit_count_days'))} ${all.length} ${peak ? ` ${esc(t('audit_peak'))} ${esc(peak.d.label)} ${String(peak.h).padStart(2, '0')}:00 · ${max}` : ''}`; return `
${_auditSummaryHeader(entries, chips)}
${dropped ? `
${esc(tf('audit_activity_capped', shown.length, all.length))}
` : ''}
${_auditActivityHeader()} ${shown.map(d => _auditActivityRow(d, max)).join('')}
${_auditActivityLegend(max)}
`; } function _auditActivityHeader() { // Every third hour is labelled: 24 numbers at this width overlap, and the grid is read by // position anyway — the exact hour of a cell is in its tooltip. let h = ''; for (let i = 0; i < 24; i++) { h += `${i % 3 === 0 ? String(i).padStart(2, '0') : ''}`; } return h + `${esc(t('audit_col_entries'))}`; } function _auditActivityRow(d, max) { let h = `${esc(d.label)}`; for (let i = 0; i < 24; i++) { const n = d.hours[i]; const title = `${d.label} ${String(i).padStart(2, '0')}:00 — ${n}`; h += ``; } return h + `${d.total}`; } /** One hue, getting darker. Both ends are theme variables, so the ramp is right in the light * and the dark theme instead of being tuned for one of them. */ function _auditHeatColor(n, max) { if (!n) return 'var(--bs-tertiary-bg)'; const pct = 22 + Math.round(78 * (max > 1 ? (n - 1) / (max - 1) : 1)); return `color-mix(in srgb, var(--bs-primary) ${pct}%, var(--bs-tertiary-bg))`; } function _auditActivityLegend(max) { const steps = [0, Math.max(1, Math.round(max / 3)), Math.max(2, Math.round(2 * max / 3)), max]; return `
${esc(t('audit_legend_less'))} ${steps.map(n => ``).join('')} ${esc(t('audit_legend_more'))}
`; }
