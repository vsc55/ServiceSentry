{# ================================================================ Audit — actors: the log summed up by WHO. The table shows every line by every actor and never a per-actor total, so a question like "did an account that nobody uses do forty things last night" can only be answered by filtering to that account — which means you have to suspect the answer before you can ask. Here every actor is one row: how much they did, how many kinds of thing, from how many addresses, and when they were first and last seen in this set. Failed logins get their own column rather than being folded into the total. A hundred entries from an admin who was working is not news; six failed logins from an account that did nothing else is, and averaged into one number they look the same. It is a SUMMARY: it describes everything the filters left standing, not the page, and it is not paginated. It also picks its own order — most active first — because "sort by time" means nothing for a row that spans a week; the sort control is hidden while it is on screen for exactly that reason. ================================================================ #} function _auditViewActors(entries) { if (!entries.length) return ''; const by = new Map(); for (const e of entries) { const who = _auditActor(e); let a = by.get(who); if (!a) { a = { who, n: 0, failed: 0, events: new Set(), ips: new Set(), first: e.ts, last: e.ts }; by.set(who, a); } a.n++; if (_auditIsFailedLogin(e)) a.failed++; a.events.add(e.event); if (e.ip) a.ips.add(e.ip); if (e.ts < a.first) a.first = e.ts; if (e.ts > a.last) a.last = e.ts; } // Most active first, then by name so equal counts keep a stable, readable order. const rows = [...by.values()].sort((x, y) => (y.n - x.n) || x.who.localeCompare(y.who)); const failing = rows.filter(r => r.failed).length; const chips = `${_summaryChip('bi-people', t('audit_count_actors'), rows.length)} ${failing ? ` ${esc(tf('audit_actors_failing', failing))}` : ''}`; return `
${_auditSummaryHeader(entries, chips)}
${rows.map(_auditActorRow).join('')}
${esc(t('col_user'))} ${esc(t('audit_col_entries'))} ${esc(t('audit_col_failed'))} ${esc(t('audit_col_kinds'))} ${esc(t('col_ip'))} ${esc(t('audit_col_first'))} ${esc(t('audit_col_last'))}
`; } function _auditActorRow(a) { // The addresses are listed, not just counted: "3 IPs" is a number, and which three is the // fact you act on. Cut to a few so one row cannot become a paragraph. const ipCell = _chipList([...a.ips], ip => _chipText(ip), 3); return ` ${esc(a.who)} ${a.n} ${a.failed ? `${a.failed}` : '—'} ${a.events.size} ${ipCell} ${esc(new Date(a.first).toLocaleString())} ${esc(new Date(a.last).toLocaleString())} `; }
